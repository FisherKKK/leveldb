- 理解系统调用开销与优化

- 掌握信号处理与中断优化

## 第1章 系统调用优化

### 1.1 系统调用开销分析

```cpp
#include <sys/time.h>
#include <unistd.h>
#include <syscall.h>

// 测量系统调用开销
void benchmark_syscall() {
    const int N = 1000000;

    // getpid系统调用
    auto start = std::chrono::high_resolution_clock::now();
    for (int i = 0; i < N; i++) {
        getpid();  // 系统调用
    }
    auto syscall_time = std::chrono::high_resolution_clock::now() - start;

    // 普通函数调用
    volatile int x = 0;
    start = std::chrono::high_resolution_clock::now();
    for (int i = 0; i < N; i++) {
        x++;  // 用户态操作
    }
    auto userspace_time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "Syscall:    " << syscall_time.count() / N << " ns\n";
    std::cout << "Userspace:  " << userspace_time.count() / N << " ns\n";
    std::cout << "Overhead:   " << (syscall_time / userspace_time) << "x\n";
}

// 典型结果：
// Syscall:    50-100 ns
// Userspace:  1-2 ns
// Overhead:   50-100x

/**
 * 系统调用开销分解：
 * 1. 模式切换（用户态→内核态）：~20-40 ns
 * 2. 保存/恢复寄存器：~10-20 ns
 * 3. 内核函数执行：~10-30 ns
 * 4. 模式切换（内核态→用户态）：~20-40 ns
 * 总计：~60-130 ns
 */
```

```cpp
#include <sys/socket.h>

// 批量发送（一次系统调用发送多个消息）
ssize_t batch_send(int sockfd, const std::vector<std::string>& messages) {
    const int batch_size = messages.size();
    std::vector<struct mmsghdr> msgvec(batch_size);
    std::vector<struct iovec> iovecs(batch_size);

    for (int i = 0; i < batch_size; i++) {
        iovecs[i].iov_base = const_cast<char*>(messages[i].data());
        iovecs[i].iov_len = messages[i].size();

        msgvec[i].msg_hdr.msg_iov = &iovecs[i];
        msgvec[i].msg_hdr.msg_iovlen = 1;
        msgvec[i].msg_hdr.msg_name = nullptr;
        msgvec[i].msg_hdr.msg_namelen = 0;
        msgvec[i].msg_hdr.msg_control = nullptr;
        msgvec[i].msg_hdr.msg_controllen = 0;
        msgvec[i].msg_hdr.msg_flags = 0;
    }

    // 一次系统调用发送所有消息
    int sent = sendmmsg(sockfd, msgvec.data(), batch_size, 0);

    return sent;
}

// 性能对比
void compare_send_methods(int sockfd) {
    std::vector<std::string> messages(1000, "Hello");

    // 方法1：逐个发送（1000次系统调用）
    auto start = std::chrono::high_resolution_clock::now();

        send(sockfd, msg.data(), msg.size(), 0);
    }
    auto single_time = std::chrono::high_resolution_clock::now() - start;

    auto batch_time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "Single send: " << single_time.count() << " ns\n";
    std::cout << "Batch send:  " << batch_time.count() << " ns\n";
```

### 2.1 CPU亲和性绑定

```cpp
#include <sched.h>
#include <pthread.h>

// 将线程绑定到特定CPU核心
void bind_to_cpu(int cpu_id) {
    cpu_set_t cpuset;
    CPU_ZERO(&cpuset);
    CPU_SET(cpu_id, &cpuset);

    pthread_t thread = pthread_self();
    int result = pthread_setaffinity_np(thread, sizeof(cpu_set_t), &cpuset);

    if (result != 0) {
        std::cerr << "Failed to set CPU affinity\n";
    }
}

// 获取当前CPU编号
int get_current_cpu() {
    return sched_getcpu();
}

// 性能对比
void benchmark_cpu_affinity() {
    const int N = 100000000;
    std::atomic<uint64_t> counter{0};

    // 无CPU绑定
    auto start = std::chrono::high_resolution_clock::now();
    for (int i = 0; i < N; i++) {
        counter.fetch_add(1, std::memory_order_relaxed);
    }
    auto no_affinity_time = std::chrono::high_resolution_clock::now() - start;

    counter = 0;

    // 有CPU绑定
    bind_to_cpu(0);
    start = std::chrono::high_resolution_clock::now();
    for (int i = 0; i < N; i++) {
        counter.fetch_add(1, std::memory_order_relaxed);
    }
    auto with_affinity_time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "No affinity:   " << no_affinity_time.count() << " ns\n";
    std::cout << "With affinity: " << with_affinity_time.count() << " ns\n";
    std::cout << "Speedup:       " << no_affinity_time / with_affinity_time << "x\n";
}
// 典型结果：1.2-2倍加速（减少Cache失效）
```

```cpp
void* numa_aware_alloc(size_t size, int node) {

}

    void* buffer = numa_aware_alloc(buffer_size, numa_node);

    // 3. 处理数据（本地访问，延迟低）

void benchmark_numa() {

    std::cout << "Remote access: " << remote_time.count() << " ns\n";
```

#### 2.2.2 NUMA感知的数据分区

```cpp
// 将数据分区到多个NUMA节点
template<typename T>
class NUMAPartitionedArray {
private:
    struct Partition {
        T* data;
        size_t size;
        int numa_node;
    };

    std::vector<Partition> partitions_;
    size_t total_size_;

public:
    NUMAPartitionedArray(size_t size) : total_size_(size) {
        int num_nodes = numa_num_configured_nodes();
        size_t partition_size = (size + num_nodes - 1) / num_nodes;

        for (int node = 0; node < num_nodes; node++) {
            size_t actual_size = std::min(partition_size,
                                          size - node * partition_size);

            if (actual_size == 0) break;

            T* data = static_cast<T*>(numa_alloc_onnode(
                actual_size * sizeof(T), node));

            partitions_.push_back({data, actual_size, node});
        }
    }

    ~NUMAPartitionedArray() {
        for (auto& part : partitions_) {
            numa_free(part.data, part.size * sizeof(T));
        }
    }

    // 获取适合当前CPU的分区
    Partition& get_local_partition() {
        int cpu = sched_getcpu();
        int node = numa_node_of_cpu(cpu);

        for (auto& part : partitions_) {
            if (part.numa_node == node) {
                return part;
            }
        }

        return partitions_[0];  // 回退
    }

    // 并行处理（每个线程处理本地分区）
    template<typename Func>
    void parallel_for_each(Func&& func) {
        std::vector<std::thread> threads;

        for (auto& part : partitions_) {
            threads.emplace_back([&part, &func]() {
                bind_to_cpu(numa_node_to_cpus(part.numa_node)[0]);

                for (size_t i = 0; i < part.size; i++) {
                    func(part.data[i]);
                }
            });
        }

        for (auto& t : threads) {
            t.join();
        }
    }
};

// 使用示例
void numa_parallel_example() {
    const size_t N = 100000000;
    NUMAPartitionedArray<int> array(N);

    // 并行初始化（每个线程访问本地内存）
    array.parallel_for_each([](int& x) {
        x = rand();
    });

    // 并行求和
    std::atomic<int64_t> total{0};
    array.parallel_for_each([&total](int& x) {
        total.fetch_add(x, std::memory_order_relaxed);
    });
}
// 性能：比单线程快2-4倍（双路NUMA）
```

### 3.1 sendfile（零拷贝文件传输）

```cpp
#include <sys/sendfile.h>

// 使用sendfile传输文件（零拷贝）
ssize_t zero_copy_transfer(int out_fd, int in_fd, off_t offset, size_t count) {
    return sendfile(out_fd, in_fd, &offset, count);
}

// 性能对比
void benchmark_sendfile() {
    const char* filename = "/tmp/testfile";
    const size_t file_size = 100 * 1024 * 1024;  // 100MB

    // 创建测试文件
    int fd = open(filename, O_CREAT | O_WRONLY, 0644);
    fallocate(fd, 0, 0, file_size);
    close(fd);

    int in_fd = open(filename, O_RDONLY);
    int out_fd = open("/dev/null", O_WRONLY);

    // 方法1：传统read/write
    auto start = std::chrono::high_resolution_clock::now();
    {
        char buffer[8192];
        ssize_t n;
        lseek(in_fd, 0, SEEK_SET);

        while ((n = read(in_fd, buffer, sizeof(buffer))) > 0) {
            write(out_fd, buffer, n);
        }
    }
    auto traditional_time = std::chrono::high_resolution_clock::now() - start;

    // 方法2：sendfile（零拷贝）
    start = std::chrono::high_resolution_clock::now();
    {
        off_t offset = 0;
        sendfile(out_fd, in_fd, &offset, file_size);
    }
    auto sendfile_time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "Traditional: " << traditional_time.count() << " ns\n";
    std::cout << "Sendfile:    " << sendfile_time.count() << " ns\n";
    std::cout << "Speedup:     " << traditional_time / sendfile_time << "x\n";

    close(in_fd);
    close(out_fd);
    unlink(filename);
}
// 典型结果：3-5倍加速

/**
 * 零拷贝原理：
 *
 * 传统方式（4次拷贝，2次系统调用）：
 * 1. DMA: 磁盘 → 内核缓冲区
 * 2. CPU: 内核缓冲区 → 用户缓冲区（read）
 * 3. CPU: 用户缓冲区 → Socket缓冲区（write）
 * 4. DMA: Socket缓冲区 → 网卡
 *
 * sendfile（2次拷贝，1次系统调用）：
 * 1. DMA: 磁盘 → 内核缓冲区
 * 2. DMA: 内核缓冲区 → 网卡（直接传输）
 */
```

```cpp
// 使用splice在两个文件描述符之间移动数据
ssize_t pipe_transfer(int in_fd, int out_fd, size_t len) {
    int pipefd[2];
    if (pipe(pipefd) < 0) {
        return -1;
    }

    // in_fd → pipe
    ssize_t bytes = splice(in_fd, nullptr, pipefd[1], nullptr, len,
                           SPLICE_F_MOVE | SPLICE_F_MORE);

    if (bytes > 0) {
        // pipe → out_fd
        splice(pipefd[0], nullptr, out_fd, nullptr, bytes,
               SPLICE_F_MOVE | SPLICE_F_MORE);
    }

    close(pipefd[0]);
    close(pipefd[1]);

    return bytes;
}
```

### 3.3 mmap（内存映射）

```cpp
#include <sys/mman.h>

// 使用mmap读取大文件
void process_large_file_mmap(const char* filename) {
    int fd = open(filename, O_RDONLY);
    if (fd < 0) {
        perror("open");
        return;
    }

    // 获取文件大小
    struct stat sb;
    fstat(fd, &sb);
    size_t file_size = sb.st_size;

    // 映射文件到内存
    void* mapped = mmap(nullptr, file_size, PROT_READ, MAP_PRIVATE, fd, 0);

    if (mapped == MAP_FAILED) {
        perror("mmap");
        close(fd);
        return;
    }

    // 提示内核顺序访问
    madvise(mapped, file_size, MADV_SEQUENTIAL);

    // 处理数据（直接访问内存）
    const char* data = static_cast<const char*>(mapped);
    for (size_t i = 0; i < file_size; i++) {
        // 处理data[i]
    }

    munmap(mapped, file_size);
    close(fd);
}

// 性能对比
void benchmark_mmap() {
    const char* filename = "/tmp/large_file";
    const size_t file_size = 1024 * 1024 * 1024;  // 1GB

    // 创建文件
    int fd = open(filename, O_CREAT | O_WRONLY, 0644);
    fallocate(fd, 0, 0, file_size);
    close(fd);

    // 方法1：read
    auto start = std::chrono::high_resolution_clock::now();
    {
        int fd = open(filename, O_RDONLY);
        char buffer[4096];
        while (read(fd, buffer, sizeof(buffer)) > 0) {}
        close(fd);
    }
    auto read_time = std::chrono::high_resolution_clock::now() - start;

    // 方法2：mmap
    start = std::chrono::high_resolution_clock::now();
    {
        int fd = open(filename, O_RDONLY);
        void* mapped = mmap(nullptr, file_size, PROT_READ, MAP_PRIVATE, fd, 0);
        madvise(mapped, file_size, MADV_SEQUENTIAL);

        volatile char sum = 0;
        const char* data = static_cast<const char*>(mapped);
        for (size_t i = 0; i < file_size; i += 4096) {
            sum += data[i];  // 触发页面加载
        }

        munmap(mapped, file_size);
        close(fd);
    }
    auto mmap_time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "read():  " << read_time.count() << " ns\n";
    std::cout << "mmap():  " << mmap_time.count() << " ns\n";
    std::cout << "Speedup: " << read_time / mmap_time << "x\n";

    unlink(filename);
}
// 典型结果：1.5-3倍加速（顺序访问大文件）
```

## 第4章 异步I/O优化

### 4.1 io_uring（现代异步I/O）

```cpp
#include <liburing.h>

// 使用io_uring的异步I/O
class IOUring {
private:
    struct io_uring ring_;
    static constexpr int QUEUE_DEPTH = 256;

public:
    IOUring() {
        int ret = io_uring_queue_init(QUEUE_DEPTH, &ring_, 0);
        if (ret < 0) {
            throw std::runtime_error("io_uring_queue_init failed");
        }
    }

    ~IOUring() {
        io_uring_queue_exit(&ring_);
    }

    // 异步读取
    void async_read(int fd, void* buffer, size_t size, off_t offset,
                    void* user_data) {
        struct io_uring_sqe* sqe = io_uring_get_sqe(&ring_);

        io_uring_prep_read(sqe, fd, buffer, size, offset);
        io_uring_sqe_set_data(sqe, user_data);

        io_uring_submit(&ring_);
    }

    // 异步写入
    void async_write(int fd, const void* buffer, size_t size, off_t offset,
                     void* user_data) {
        struct io_uring_sqe* sqe = io_uring_get_sqe(&ring_);

        io_uring_prep_write(sqe, fd, buffer, size, offset);
        io_uring_sqe_set_data(sqe, user_data);

        io_uring_submit(&ring_);
    }

    // 等待完成
    int wait_completion(void** user_data) {
        struct io_uring_cqe* cqe;
        int ret = io_uring_wait_cqe(&ring_, &cqe);

        if (ret < 0) {
            return ret;
        }

        *user_data = io_uring_cqe_get_data(cqe);
        int result = cqe->res;

        io_uring_cqe_seen(&ring_, cqe);

        return result;
    }

    // 批量提交
    int submit_batch() {
        return io_uring_submit(&ring_);
    }
};

// 使用示例：并发读取多个文件
void benchmark_io_uring() {
    const int NUM_FILES = 100;
    const size_t BLOCK_SIZE = 4096;

    std::vector<int> fds;
    std::vector<std::vector<char>> buffers(NUM_FILES,
                                            std::vector<char>(BLOCK_SIZE));

    // 创建测试文件
    for (int i = 0; i < NUM_FILES; i++) {
        std::string filename = "/tmp/test_" + std::to_string(i);
        int fd = open(filename.c_str(), O_CREAT | O_RDWR, 0644);
        fallocate(fd, 0, 0, BLOCK_SIZE);
        fds.push_back(fd);
    }

    // 方法1：同步I/O
    auto start = std::chrono::high_resolution_clock::now();
    for (int i = 0; i < NUM_FILES; i++) {
        read(fds[i], buffers[i].data(), BLOCK_SIZE);
    }
    auto sync_time = std::chrono::high_resolution_clock::now() - start;

    // 方法2：io_uring异步I/O
    IOUring uring;

    start = std::chrono::high_resolution_clock::now();

    // 提交所有读取请求
    for (int i = 0; i < NUM_FILES; i++) {
        uring.async_read(fds[i], buffers[i].data(), BLOCK_SIZE, 0,
                         reinterpret_cast<void*>(i));
    }

    // 等待所有完成
    for (int i = 0; i < NUM_FILES; i++) {
        void* user_data;
        uring.wait_completion(&user_data);
    }

    auto async_time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "Sync I/O:   " << sync_time.count() << " ns\n";
    std::cout << "io_uring:   " << async_time.count() << " ns\n";
    std::cout << "Speedup:    " << sync_time / async_time << "x\n";

    // 清理
    for (int i = 0; i < NUM_FILES; i++) {
        close(fds[i]);
        unlink(("/tmp/test_" + std::to_string(i)).c_str());
    }
}
// 典型结果：2-5倍加速
```

```cpp
    struct Request {
        struct aiocb cb;
        std::function<void(ssize_t)> callback;
    };

    void async_read(int fd, void* buffer, size_t size, off_t offset,
                    std::function<void(ssize_t)> callback) {
        Request* req = new Request();

        memset(&req->cb, 0, sizeof(struct aiocb));
        req->cb.aio_fildes = fd;
        req->cb.aio_buf = buffer;
        req->cb.aio_nbytes = size;
        req->cb.aio_offset = offset;
        req->cb.aio_sigevent.sigev_notify = SIGEV_NONE;

        req->callback = callback;

        int ret = aio_read(&req->cb);
        if (ret < 0) {
            delete req;
            throw std::runtime_error("aio_read failed");
        }
    }

    void wait_all() {
        // 等待所有异步操作完成
        // 简化实现
    }
};
```

### 5.1 Direct I/O（绕过Page Cache）

```cpp
#include <fcntl.h>

// Direct I/O示例（数据库常用）
void direct_io_example() {
    const size_t BLOCK_SIZE = 4096;
    const size_t NUM_BLOCKS = 1000;

    // 打开文件（O_DIRECT标志）
    int fd = open("/tmp/direct_io_test",
                  O_CREAT | O_RDWR | O_DIRECT,
                  0644);

    if (fd < 0) {
        perror("open with O_DIRECT");
        return;
    }

    // Direct I/O要求：
    // 1. 缓冲区地址对齐到512字节
    // 2. 读写大小是512字节的倍数
    // 3. 文件偏移是512字节的倍数

    void* buffer;
    posix_memalign(&buffer, 512, BLOCK_SIZE);  // 对齐分配

    // 写入
    for (size_t i = 0; i < NUM_BLOCKS; i++) {
        ssize_t written = write(fd, buffer, BLOCK_SIZE);
        if (written != BLOCK_SIZE) {
            perror("write");
            break;
        }
    }

    // 同步到磁盘
    fsync(fd);

    free(buffer);
    close(fd);
    unlink("/tmp/direct_io_test");
}

// 性能对比
void benchmark_direct_io() {
    const size_t BLOCK_SIZE = 4096;
    const size_t NUM_BLOCKS = 100000;  // 400MB

    void* buffer;
    posix_memalign(&buffer, 512, BLOCK_SIZE);
    memset(buffer, 0, BLOCK_SIZE);

    // 方法1：Buffered I/O
    auto start = std::chrono::high_resolution_clock::now();
    {
        int fd = open("/tmp/buffered_test", O_CREAT | O_WRONLY, 0644);

        for (size_t i = 0; i < NUM_BLOCKS; i++) {
            write(fd, buffer, BLOCK_SIZE);
        }

        fsync(fd);
        close(fd);
        unlink("/tmp/buffered_test");
    }
    auto buffered_time = std::chrono::high_resolution_clock::now() - start;

    // 方法2：Direct I/O
    start = std::chrono::high_resolution_clock::now();
    {
        int fd = open("/tmp/direct_test",
                      O_CREAT | O_WRONLY | O_DIRECT,
                      0644);

        for (size_t i = 0; i < NUM_BLOCKS; i++) {
            write(fd, buffer, BLOCK_SIZE);
        }

        fsync(fd);
        close(fd);
        unlink("/tmp/direct_test");
    }
    auto direct_time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "Buffered I/O: " << buffered_time.count() << " ns\n";
    std::cout << "Direct I/O:   " << direct_time.count() << " ns\n";
```

### 6.1 信号处理优化

```cpp
#include <signal.h>

// 高性能信号处理：使用signalfd
int create_signal_fd() {
    sigset_t mask;
    sigemptyset(&mask);
    sigaddset(&mask, SIGINT);
    sigaddset(&mask, SIGTERM);

    // 阻塞信号
    sigprocmask(SIG_BLOCK, &mask, nullptr);

    // 创建signalfd
    int sfd = signalfd(-1, &mask, SFD_NONBLOCK | SFD_CLOEXEC);

    return sfd;
}

void handle_signals(int sfd) {
    struct signalfd_siginfo si;

    while (read(sfd, &si, sizeof(si)) == sizeof(si)) {
        if (si.ssi_signo == SIGINT) {
            std::cout << "Received SIGINT\n";
        } else if (si.ssi_signo == SIGTERM) {
            std::cout << "Received SIGTERM\n";
        }
    }
}

// vs 传统signal/sigaction: 更高效，可与epoll集成
```

### 7.1 DPDK（用户态网络栈）

```cpp
/**
 * DPDK原理：
 * 1. 绕过内核协议栈
 * 2. 用户态直接访问网卡
 * 3. 轮询模式（避免中断）
 * 4. 零拷贝（DMA直接到用户内存）
 *
 * 性能提升：
 * - 延迟：内核栈~50us → DPDK~1us（50倍）
 * - 吞吐量：内核栈~1Gbps → DPDK~10-100Gbps（10-100倍）
 */

// 简化的DPDK示例（伪代码）
void dpdk_packet_processing() {
    // 初始化DPDK
    rte_eal_init(argc, argv);

    // 配置网卡
    rte_eth_dev_configure(port_id, rx_queues, tx_queues, &port_conf);

    // 分配内存池
    mbuf_pool = rte_pktmbuf_pool_create("MBUF_POOL", NUM_MBUFS,
                                        MBUF_CACHE_SIZE, 0,
                                        RTE_MBUF_DEFAULT_BUF_SIZE,
                                        rte_socket_id());

    // 主循环：轮询接收数据包
    while (running) {
        // 批量接收（0拷贝）
        struct rte_mbuf* bufs[BURST_SIZE];
        uint16_t nb_rx = rte_eth_rx_burst(port_id, 0, bufs, BURST_SIZE);

        // 处理数据包
        for (uint16_t i = 0; i < nb_rx; i++) {
            process_packet(bufs[i]);
        }

        // 批量发送
        uint16_t nb_tx = rte_eth_tx_burst(port_id, 0, bufs, nb_rx);

        // 释放未发送的包
        for (uint16_t i = nb_tx; i < nb_rx; i++) {
            rte_pktmbuf_free(bufs[i]);
        }
    }
}

// 适用场景：
// - 高频交易
// - 网络功能虚拟化（NFV）
// - 5G基站
// - CDN边缘节点
```

## 总结

系统级性能优化关键技术：

| 优化技术 | 性能提升 | 适用场景 |
|---------|---------|---------|
| 批量系统调用 | 10-50x | 网络I/O, 文件I/O |
| CPU亲和性 | 1.2-2x | 多线程应用 |
| NUMA优化 | 1.5-3x | 双路/多路服务器 |
| 零拷贝 | 3-10x | 文件传输 |
| io_uring | 2-5x | 异步I/O |
| Direct I/O | 1-2x | 数据库 |
| DPDK | 10-100x | 高性能网络 |
| SPDK | 5-10x | 高性能存储 |

**优化原则：**
1. **减少系统调用**：批量操作、缓存结果
2. **减少数据拷贝**：零拷贝、DMA、mmap
3. **减少上下文切换**：CPU绑定、减少线程数
4. **利用硬件特性**：NUMA、DMA、硬件卸载
