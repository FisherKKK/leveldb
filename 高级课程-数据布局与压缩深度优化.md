**Cache Line层级：**
```

L2 Cache: 256KB, 4-way, 64字节/line, 延迟 ~12周期
```

### 1.2 False Sharing问题

```cpp
#include <thread>
#include <atomic>
#include <chrono>
#include <vector>

// 不好：False Sharing
struct BadCounter {
    std::atomic<int> count1{0};  // 假设在同一cache line
    std::atomic<int> count2{0};
    std::atomic<int> count3{0};
    std::atomic<int> count4{0};
};

// 好：Cache Line对齐
struct alignas(64) GoodCounter {
    std::atomic<int> count{0};
    char padding[60];  // 填充到64字节
};

// 或使用C++17
struct GoodCounter2 {
    alignas(std::hardware_destructive_interference_size) std::atomic<int> count{0};
};

void benchmark_false_sharing() {
    const int iterations = 10000000;
    const int num_threads = 4;

    // 测试1：False Sharing
    BadCounter bad;
    auto start = std::chrono::high_resolution_clock::now();

    std::vector<std::thread> threads;
    threads.emplace_back([&]() {
        for (int i = 0; i < iterations; i++) bad.count1++;
    });
    threads.emplace_back([&]() {
        for (int i = 0; i < iterations; i++) bad.count2++;
    });
    threads.emplace_back([&]() {
        for (int i = 0; i < iterations; i++) bad.count3++;
    });
    threads.emplace_back([&]() {
        for (int i = 0; i < iterations; i++) bad.count4++;
    });

    for (auto& t : threads) t.join();
    auto bad_time = std::chrono::high_resolution_clock::now() - start;

    // 测试2：无False Sharing
    std::array<GoodCounter, 4> good;
    start = std::chrono::high_resolution_clock::now();

    threads.clear();
    for (int i = 0; i < num_threads; i++) {
        threads.emplace_back([&, i]() {
            for (int j = 0; j < iterations; j++) {
                good[i].count++;
            }
        });
    }

    for (auto& t : threads) t.join();
    auto good_time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "With False Sharing:    "
              << std::chrono::duration_cast<std::chrono::milliseconds>(bad_time).count()
              << " ms\n";
    std::cout << "Without False Sharing: "
              << std::chrono::duration_cast<std::chrono::milliseconds>(good_time).count()
              << " ms\n";
    std::cout << "Speedup: "
              << (double)bad_time.count() / good_time.count() << "x\n";
}

// 典型输出：
// With False Sharing:    3850 ms
// Without False Sharing: 620 ms
// Speedup: 6.2x
```

### 1.3 Cache-Friendly数组设计

```cpp
// 不好：指针跳转
struct Node {
    int data;
    Node* next;
};

void bad_linked_list_sum(Node* head) {
    int sum = 0;
    for (Node* p = head; p != nullptr; p = p->next) {
        sum += p->data;  // 每次都是cache miss
    }
}

// 好：连续内存
void good_array_sum(const std::vector<int>& data) {
    int sum = 0;
    for (int val : data) {
        sum += val;  // cache预取友好
    }
}

// 性能对比
void benchmark_layout() {
    const int N = 1000000;

    // 链表版本
    Node* head = new Node{0, nullptr};
    Node* current = head;
    for (int i = 1; i < N; i++) {
        current->next = new Node{i, nullptr};
        current = current->next;
    }

    auto start = std::chrono::high_resolution_clock::now();
    bad_linked_list_sum(head);
    auto list_time = std::chrono::high_resolution_clock::now() - start;

    // 数组版本
    std::vector<int> arr(N);
    for (int i = 0; i < N; i++) arr[i] = i;

    start = std::chrono::high_resolution_clock::now();
    good_array_sum(arr);
    auto array_time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "Linked List: " << list_time.count() << " ns\n";
    std::cout << "Array:       " << array_time.count() << " ns\n";
    std::cout << "Speedup:     " << (double)list_time.count() / array_time.count() << "x\n";

    // 清理
    while (head) {
        Node* tmp = head;
        head = head->next;
        delete tmp;
    }
}

// 典型输出：
// Linked List: 45000000 ns
// Array:       5000000 ns
// Speedup:     9x
```

### 2.3 Hot/Cold数据分离

```cpp
// 不好：混合hot/cold数据
struct BadEntity {
    // Hot: 频繁访问
    float position_x, position_y, position_z;
    float velocity_x, velocity_y, velocity_z;

    // Cold: 很少访问
    std::string name;
    std::string description;
    uint64_t creation_time;
    uint64_t last_modified;
};

// 好：分离hot/cold数据
struct EntityCore {  // Hot data - 单独cache line
    alignas(64) float position_x, position_y, position_z;
    float velocity_x, velocity_y, velocity_z;
    uint32_t id;
    // 24字节，剩余40字节padding
};

struct EntityMetadata {  // Cold data
    std::string name;
    std::string description;
    uint64_t creation_time;
    uint64_t last_modified;
    uint32_t core_id;  // 指向EntityCore的索引
};

// 性能对比
void benchmark_hot_cold_separation() {
    const int N = 100000;
    const int iterations = 1000;

    // 测试1：混合布局
    std::vector<BadEntity> bad_entities(N);
    auto start = std::chrono::high_resolution_clock::now();

    for (int iter = 0; iter < iterations; iter++) {
        for (auto& e : bad_entities) {
            e.position_x += e.velocity_x * 0.016f;  // 只访问hot数据
            e.position_y += e.velocity_y * 0.016f;
            e.position_z += e.velocity_z * 0.016f;
        }
    }

    auto bad_time = std::chrono::high_resolution_clock::now() - start;

    // 测试2：分离布局
    std::vector<EntityCore> cores(N);
    std::vector<EntityMetadata> metadata(N);

    start = std::chrono::high_resolution_clock::now();

    for (int iter = 0; iter < iterations; iter++) {
        for (auto& e : cores) {
            e.position_x += e.velocity_x * 0.016f;
            e.position_y += e.velocity_y * 0.016f;
            e.position_z += e.velocity_z * 0.016f;
        }
    }

    auto good_time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "Mixed Layout:     " << bad_time.count() << " ns\n";
    std::cout << "Separated Layout: " << good_time.count() << " ns\n";
    std::cout << "Speedup:          "
              << (double)bad_time.count() / good_time.count() << "x\n";
}

// 典型输出：
// Mixed Layout:     850ms
// Separated Layout: 320ms
// Speedup:          2.65x
```

## 3. AoS vs SoA

### 3.1 Array of Structures (AoS)

```cpp
// AoS: 传统布局
struct Particle {
    float x, y, z;      // 位置
    float vx, vy, vz;   // 速度
};

std::vector<Particle> particles_aos(1000000);

void update_aos(std::vector<Particle>& particles, float dt) {
    for (auto& p : particles) {
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.z += p.vz * dt;
    }
}
```

### 3.2 Structure of Arrays (SoA)

```cpp
// SoA: SIMD友好布局
struct ParticlesSoA {
    std::vector<float> x, y, z;
    std::vector<float> vx, vy, vz;

    ParticlesSoA(size_t n) {
        x.resize(n); y.resize(n); z.resize(n);
        vx.resize(n); vy.resize(n); vz.resize(n);
    }
};

void update_soa(ParticlesSoA& particles, float dt) {
    for (size_t i = 0; i < particles.x.size(); i++) {
        particles.x[i] += particles.vx[i] * dt;
        particles.y[i] += particles.vy[i] * dt;
        particles.z[i] += particles.vz[i] * dt;
    }
}

// SIMD版本
#include <immintrin.h>

void update_soa_simd(ParticlesSoA& particles, float dt) {
    __m256 dt_vec = _mm256_set1_ps(dt);

    size_t i = 0;
    for (; i + 8 <= particles.x.size(); i += 8) {
        // 加载8个粒子的x坐标
        __m256 x = _mm256_loadu_ps(&particles.x[i]);
        __m256 vx = _mm256_loadu_ps(&particles.vx[i]);

        // x += vx * dt
        x = _mm256_fmadd_ps(vx, dt_vec, x);

        // 存储
        _mm256_storeu_ps(&particles.x[i], x);

        // y和z同理...
    }

    // 处理剩余元素
    for (; i < particles.x.size(); i++) {
        particles.x[i] += particles.vx[i] * dt;
    }
}
```


### 5.2 LevelDB中的Snappy使用

```cpp
// leveldb/table/format.cc中的压缩逻辑
namespace leveldb {

// 压缩Block
void CompressBlock(const Slice& raw, CompressionType type, std::string* compressed) {
    switch (type) {
        case kNoCompression:
            compressed->assign(raw.data(), raw.size());
            break;

        case kSnappyCompression: {
            std::string buf;
            snappy::Compress(raw.data(), raw.size(), &buf);

            // 只有压缩有效才使用
            if (buf.size() < raw.size() - (raw.size() / 8)) {
                *compressed = std::move(buf);
            } else {
                // 压缩无效，使用原始数据
                compressed->assign(raw.data(), raw.size());
            }
            break;
        }
    }
}

// 解压Block
Status UncompressBlock(const Slice& compressed, CompressionType type,
                       BlockContents* result) {
    switch (type) {
        case kNoCompression:
            result->data = Slice(compressed.data(), compressed.size());
            result->heap_allocated = false;
            result->cachable = false;
            return Status::OK();

        case kSnappyCompression: {
            size_t ulength = 0;
            if (!snappy::GetUncompressedLength(compressed.data(),
                                               compressed.size(),
                                               &ulength)) {
                return Status::Corruption("corrupted compressed block");
            }

            char* ubuf = new char[ulength];
            if (!snappy::RawUncompress(compressed.data(),
                                       compressed.size(),
                                       ubuf)) {
                delete[] ubuf;
                return Status::Corruption("corrupted compressed block");
            }

            result->data = Slice(ubuf, ulength);
            result->heap_allocated = true;
            result->cachable = true;
            return Status::OK();
        }
    }
}

}  // namespace leveldb
```

### 6.2 Zstd高级特性：字典压缩

```cpp
// 字典压缩：适合大量小对象
class ZstdDictCompressor {
private:
    ZSTD_CDict* cdict_;
    ZSTD_DDict* ddict_;

public:
    // 训练字典
    ZstdDictCompressor(const std::vector<std::string>& samples, size_t dict_size = 112640) {
        // 准备训练数据
        std::string training_data;
        std::vector<size_t> sizes;

        for (const auto& sample : samples) {
            training_data += sample;
            sizes.push_back(sample.size());
        }

        // 训练字典
        std::vector<char> dict(dict_size);
        size_t trained_size = ZDICT_trainFromBuffer(
            dict.data(), dict.size(),
            training_data.data(), sizes.data(), sizes.size()
        );

        dict.resize(trained_size);

        // 创建压缩/解压字典
        cdict_ = ZSTD_createCDict(dict.data(), dict.size(), 3);
        ddict_ = ZSTD_createDDict(dict.data(), dict.size());
    }

    ~ZstdDictCompressor() {
        ZSTD_freeCDict(cdict_);
        ZSTD_freeDDict(ddict_);
    }

    std::string compress(const std::string& input) {
        size_t bound = ZSTD_compressBound(input.size());
        std::string compressed(bound, '\0');

        ZSTD_CCtx* cctx = ZSTD_createCCtx();
        size_t size = ZSTD_compress_usingCDict(
            cctx,
            compressed.data(), compressed.size(),
            input.data(), input.size(),
            cdict_
        );
        ZSTD_freeCCtx(cctx);

        compressed.resize(size);
        return compressed;
    }

    std::string decompress(const std::string& compressed, size_t original_size) {
        std::string decompressed(original_size, '\0');

        ZSTD_DCtx* dctx = ZSTD_createDCtx();
        ZSTD_decompress_usingDDict(
            dctx,
            decompressed.data(), decompressed.size(),
            compressed.data(), compressed.size(),
            ddict_
        );
        ZSTD_freeDCtx(dctx);

        return decompressed;
    }
};
```

## 9. VarInt变长编码

### 9.1 VarInt原理

```cpp
// VarInt: 变长整数编码
// 小数字用少量字节，大数字用更多字节
//
// 编码规则（Protocol Buffers风格）：
// - 每字节最高位：1=还有后续字节，0=最后一个字节
// - 每字节低7位：数据位
//
// 示例：
// 300 = 0b100101100
//     = 0b10 0101100
//     = [10101100 00000010]  (低字节在前)
//     = [0xAC, 0x02]

#include <vector>
#include <cstdint>

class VarInt {
public:
    // 编码
    static void encode(uint64_t value, std::vector<uint8_t>& output) {
        while (value >= 0x80) {
            output.push_back((value & 0x7F) | 0x80);
            value >>= 7;
        }
        output.push_back(value & 0x7F);
    }

    // 解码
    static uint64_t decode(const uint8_t* data, size_t& bytes_read) {
        uint64_t result = 0;
        int shift = 0;
        bytes_read = 0;

        while (true) {
            uint8_t byte = data[bytes_read++];
            result |= uint64_t(byte & 0x7F) << shift;

            if ((byte & 0x80) == 0) {
                break;
            }

            shift += 7;
        }

        return result;
    }

    // 计算编码大小
    static size_t encoded_size(uint64_t value) {
        size_t size = 1;
        while (value >= 0x80) {
            size++;
            value >>= 7;
        }
        return size;
    }
};

void varint_example() {
    std::vector<uint64_t> numbers = {0, 127, 128, 255, 256, 16384, 1000000, UINT64_MAX};

    std::cout << "Number      | Fixed(8B) | VarInt | Savings\n";
    std::cout << "------------|-----------|--------|--------\n";

    for (uint64_t num : numbers) {
        std::vector<uint8_t> encoded;
        VarInt::encode(num, encoded);

        printf("%11llu | %9zu | %6zu | %6.1f%%\n",
               num, sizeof(uint64_t), encoded.size(),
               (1.0 - (double)encoded.size() / sizeof(uint64_t)) * 100);
    }
}

// 输出：
// Number      | Fixed(8B) | VarInt | Savings

//         256 |         8 |      2 |   75.0%
```

```cpp
namespace leveldb {

// 编码32位VarInt
inline void EncodeVarint32(std::string* dst, uint32_t v) {
    char buf[5];
    char* ptr = EncodeVarint32(buf, v);
    dst->append(buf, ptr - buf);
}

inline char* EncodeVarint32(char* dst, uint32_t v) {
    uint8_t* ptr = reinterpret_cast<uint8_t*>(dst);
    static const int B = 128;
    if (v < (1 << 7)) {
        *(ptr++) = v;
    } else if (v < (1 << 14)) {
        *(ptr++) = v | B;
        *(ptr++) = v >> 7;
    } else if (v < (1 << 21)) {
        *(ptr++) = v | B;
        *(ptr++) = (v >> 7) | B;
        *(ptr++) = v >> 14;
    } else if (v < (1 << 28)) {
        *(ptr++) = v | B;
        *(ptr++) = (v >> 7) | B;
        *(ptr++) = (v >> 14) | B;
        *(ptr++) = v >> 21;
    } else {
        *(ptr++) = v | B;
        *(ptr++) = (v >> 7) | B;
        *(ptr++) = (v >> 14) | B;
        *(ptr++) = (v >> 21) | B;
        *(ptr++) = v >> 28;
    }
    return reinterpret_cast<char*>(ptr);
}

// 解码
inline bool GetVarint32(Slice* input, uint32_t* value) {
    const char* p = input->data();
    const char* limit = p + input->size();
    const char* q = GetVarint32Ptr(p, limit, value);
    if (q == nullptr) {
        return false;
    } else {
        *input = Slice(q, limit - q);
        return true;
    }
}
```

## 10. Delta编码

### 10.1 Delta编码原理

```cpp
// Delta编码：存储差值而不是绝对值
// 适用于：时间序列、递增ID、有序key等

class DeltaEncoder {
public:
    // 编码
    static std::vector<int32_t> encode(const std::vector<int64_t>& values) {
        if (values.empty()) return {};

        std::vector<int32_t> deltas;
        deltas.reserve(values.size());

        int64_t prev = 0;
        for (int64_t value : values) {
            deltas.push_back(static_cast<int32_t>(value - prev));
            prev = value;
        }

        return deltas;
    }

    // 解码
    static std::vector<int64_t> decode(const std::vector<int32_t>& deltas) {
        std::vector<int64_t> values;
        values.reserve(deltas.size());

        int64_t sum = 0;
        for (int32_t delta : deltas) {
            sum += delta;
            values.push_back(sum);
        }

        return values;
    }
};

void delta_encoding_example() {
    // 递增ID序列
    std::vector<int64_t> ids = {
        1000, 1001, 1002, 1003, 1005, 1008, 1010, 1015
    };

    // Delta编码
    auto deltas = DeltaEncoder::encode(ids);

    // 计算大小
    size_t original_size = ids.size() * sizeof(int64_t);

    // Delta + VarInt
    size_t encoded_size = 0;
    for (int32_t delta : deltas) {
        encoded_size += VarInt::encoded_size(delta);
    }

    std::cout << "Original IDs:  ";
    for (auto id : ids) std::cout << id << " ";
    std::cout << "\nDeltas:        ";
    for (auto d : deltas) std::cout << d << " ";
    std::cout << "\n\n";

    std::cout << "Original size: " << original_size << " bytes\n";
    std::cout << "Encoded size:  " << encoded_size << " bytes\n";
    std::cout << "Compression:   " << (double)original_size / encoded_size << "x\n";
}

// 输出：
// Original IDs:  1000 1001 1002 1003 1005 1008 1010 1015
// Deltas:        1000 1 1 1 2 3 2 5
//
// Original size: 64 bytes
// Encoded size:  11 bytes
// Compression:   5.8x
```

### 10.2 Delta-of-Delta编码（时间序列）

```cpp
// Delta-of-Delta: 对delta再次delta
// 适用于：均匀递增的时间序列

class DoubleDeltaEncoder {
public:
    static std::vector<int16_t> encode(const std::vector<int64_t>& timestamps) {
        if (timestamps.size() < 2) return {};

        std::vector<int16_t> double_deltas;
        double_deltas.reserve(timestamps.size());

        int64_t prev_delta = timestamps[1] - timestamps[0];
        double_deltas.push_back(0);  // 第一个delta没有previous

        for (size_t i = 2; i < timestamps.size(); i++) {
            int64_t delta = timestamps[i] - timestamps[i-1];
            int64_t double_delta = delta - prev_delta;
            double_deltas.push_back(static_cast<int16_t>(double_delta));
            prev_delta = delta;
        }

        return double_deltas;
    }
};

void double_delta_example() {
    // 每秒递增的时间戳
    std::vector<int64_t> timestamps;
    int64_t base = 1700000000;
    for (int i = 0; i < 100; i++) {
        timestamps.push_back(base + i);  // 均匀递增
    }

    // 添加一些噪声
    timestamps[50] += 2;
    timestamps[75] -= 1;

    auto double_deltas = DoubleDeltaEncoder::encode(timestamps);

    // 统计double delta的分布
    std::map<int16_t, int> distribution;
    for (auto dd : double_deltas) {
        distribution[dd]++;
    }

    std::cout << "Double Delta Distribution:\n";
    for (const auto& [value, count] : distribution) {
        std::cout << value << ": " << count << " times\n";
    }

    // 大部分double delta是0，可以进一步压缩
    size_t zeros = distribution[0];
    std::cout << "\nZero double deltas: " << zeros << "/" << double_deltas.size()
              << " (" << (double)zeros / double_deltas.size() * 100 << "%)\n";
}

// 输出：
// Double Delta Distribution:
// -1: 1 times
// 0: 95 times
// 1: 1 times
// 2: 1 times
//
// Zero double deltas: 95/98 (96.9%)
```

## 13. LevelDB SSTable格式

### 13.1 SSTable结构

```cpp
// LevelDB SSTable格式：
//
// [Data Block 1]
// [Data Block 2]
// ...
// [Data Block N]
// [Meta Block 1: filter block]
// [Meta Block 2: stats]
// ...
// [Meta Index Block]  -- 指向meta blocks
// [Index Block]       -- 指向data blocks
// [Footer]            -- 指向meta index和index block
//
// Footer格式（固定48字节）：
// [metaindex_handle: varint64 offset + varint64 size]
// [index_handle: varint64 offset + varint64 size]
// [padding]
// [magic number: 0xdb4775248b80fb57]

namespace leveldb {

// Block Handle: 指向block的位置
class BlockHandle {
private:
    uint64_t offset_;
    uint64_t size_;

public:
    void EncodeTo(std::string* dst) const {
        PutVarint64(dst, offset_);
        PutVarint64(dst, size_);
    }

    Status DecodeFrom(Slice* input) {
        if (GetVarint64(input, &offset_) &&
            GetVarint64(input, &size_)) {
            return Status::OK();
        } else {
            return Status::Corruption("bad block handle");
        }
    }
};

// Footer: 文件末尾的固定格式
class Footer {
private:
    BlockHandle metaindex_handle_;
    BlockHandle index_handle_;

public:
    static constexpr size_t kEncodedLength = 2 * BlockHandle::kMaxEncodedLength + 8;

    void EncodeTo(std::string* dst) const {
        const size_t original_size = dst->size();
        metaindex_handle_.EncodeTo(dst);
        index_handle_.EncodeTo(dst);
        dst->resize(kEncodedLength + original_size);  // 填充到固定大小
        PutFixed64(dst, kTableMagicNumber);  // Magic number
    }

    Status DecodeFrom(Slice* input) {
        if (input->size() < kEncodedLength) {
            return Status::Corruption("not an sstable (footer too short)");
        }

        // 检查magic number
        const char* magic_ptr = input->data() + kEncodedLength - 8;
        const uint64_t magic = DecodeFixed64(magic_ptr);
        if (magic != kTableMagicNumber) {
            return Status::Corruption("not an sstable (bad magic number)");
        }

        Status result = metaindex_handle_.DecodeFrom(input);
        if (result.ok()) {
            result = index_handle_.DecodeFrom(input);
        }
        return result;
    }
};

}  // namespace leveldb
```

### 13.2 Data Block格式

```cpp
// Data Block结构：
// [record 0]
// [record 1]
// ...
// [record N-1]
// [restart point 0: uint32]
// [restart point 1: uint32]
// ...
// [restart point M-1: uint32]
// [num restart points: uint32]
//
// Record格式（前缀压缩）：
// [shared_bytes: varint32]   -- 与前一个key的共享前缀长度
// [unshared_bytes: varint32] -- 非共享部分长度
// [value_length: varint32]
// [unshared_key_data]
// [value_data]

namespace leveldb {

class BlockBuilder {
private:
    const Options* options_;
    std::string buffer_;               // 数据buffer
    std::vector<uint32_t> restarts_;   // Restart点
    int counter_;                       // 自上次restart的record数
    bool finished_;
    std::string last_key_;              // 上一个key

public:
    explicit BlockBuilder(const Options* options)
        : options_(options), finished_(false), counter_(0) {
        restarts_.push_back(0);  // 第一个restart点
    }

    void Add(const Slice& key, const Slice& value) {
        Slice last_key_piece(last_key_);
        assert(!finished_);
        assert(counter_ <= options_->block_restart_interval);
        assert(buffer_.empty() || key > last_key_piece);

        size_t shared = 0;
        if (counter_ < options_->block_restart_interval) {
            // 计算共享前缀
            const size_t min_length = std::min(last_key_piece.size(), key.size());
            while ((shared < min_length) && (last_key_piece[shared] == key[shared])) {
                shared++;
            }
        } else {
            // 达到restart interval，添加restart point
            restarts_.push_back(buffer_.size());
            counter_ = 0;
        }

        const size_t non_shared = key.size() - shared;

        // 编码：<shared><non_shared><value_size>
        PutVarint32(&buffer_, shared);
        PutVarint32(&buffer_, non_shared);
        PutVarint32(&buffer_, value.size());

        // 添加非共享key和value
        buffer_.append(key.data() + shared, non_shared);
        buffer_.append(value.data(), value.size());

        // 更新状态
        last_key_.resize(shared);
        last_key_.append(key.data() + shared, non_shared);
        counter_++;
    }

    Slice Finish() {
        // 添加restart points
        for (uint32_t restart : restarts_) {
            PutFixed32(&buffer_, restart);
        }
        PutFixed32(&buffer_, restarts_.size());
        finished_ = true;
        return Slice(buffer_);
    }
};

}  // namespace leveldb
```

### 13.3 前缀压缩示例

```cpp
void prefix_compression_example() {
    std::vector<std::pair<std::string, std::string>> entries = {
        {"user:1000:name", "Alice"},
        {"user:1000:email", "alice@example.com"},
        {"user:1001:name", "Bob"},
        {"user:1001:email", "bob@example.com"},
    };

    // 计算未压缩大小
    size_t uncompressed = 0;
    for (const auto& [key, value] : entries) {
        uncompressed += key.size() + value.size();
    }

    // 模拟前缀压缩
    size_t compressed = 0;
    std::string prev_key;
    for (const auto& [key, value] : entries) {
        // 计算共享前缀
        size_t shared = 0;
        size_t min_len = std::min(prev_key.size(), key.size());
        while (shared < min_len && prev_key[shared] == key[shared]) {
            shared++;
        }

        size_t non_shared = key.size() - shared;

        // VarInt编码开销 + 非共享key + value
        compressed += VarInt::encoded_size(shared);
        compressed += VarInt::encoded_size(non_shared);
        compressed += VarInt::encoded_size(value.size());
        compressed += non_shared;
        compressed += value.size();

        prev_key = key;

        std::cout << "Key: " << key << "\n";

    std::cout << "Compressed:   " << compressed << " bytes\n";
```

### 15.1 高性能KV存储

```cpp
// 综合应用：Cache-Friendly + 压缩 + 编码

class HighPerfKVStore {
private:
    // SoA布局：Cache友好
    struct KeyValueSoA {
        std::vector<uint32_t> key_offsets;    // Key在buffer中的偏移
        std::vector<uint32_t> key_lengths;
        std::vector<uint32_t> value_offsets;
        std::vector<uint32_t> value_lengths;
        std::string data_buffer;               // 压缩后的数据
    };

    KeyValueSoA data_;
    ZstdDictCompressor compressor_;

public:
    // 批量插入（优化写入）
    void batch_insert(const std::vector<std::pair<std::string, std::string>>& kvs) {
        // 1. Delta编码keys（如果是递增ID）
        // 2. 字典压缩values
        // 3. SoA布局存储

        for (const auto& [key, value] : kvs) {
            // 压缩value
            std::string compressed_value = compressor_.compress(value);

            // 记录偏移和长度
            data_.key_offsets.push_back(data_.data_buffer.size());
            data_.key_lengths.push_back(key.size());
            data_.data_buffer.append(key);

            data_.value_offsets.push_back(data_.data_buffer.size());
            data_.value_lengths.push_back(compressed_value.size());
            data_.data_buffer.append(compressed_value);
        }
    }

    // 范围查询（Cache友好遍历）
    std::vector<std::string> range_query(size_t start, size_t end) {
        std::vector<std::string> results;

        for (size_t i = start; i < end && i < data_.key_offsets.size(); i++) {
            // 顺序访问：预取友好
            uint32_t value_offset = data_.value_offsets[i];
            uint32_t value_length = data_.value_lengths[i];

            std::string compressed_value =
                data_.data_buffer.substr(value_offset, value_length);

            // 解压
            std::string value = compressor_.decompress(
                compressed_value,
                /* 需要原始大小信息 */ 1024
            );

            results.push_back(std::move(value));
        }

        return results;
    }

    // 统计信息
    void print_stats() const {
        size_t overhead = data_.key_offsets.size() * sizeof(uint32_t) * 4;
        size_t total = overhead + data_.data_buffer.size();

        std::cout << "Entries:      " << data_.key_offsets.size() << "\n";
        std::cout << "Data size:    " << data_.data_buffer.size() << " bytes\n";
        std::cout << "Overhead:     " << overhead << " bytes\n";
        std::cout << "Total:        " << total << " bytes\n";
        std::cout << "Bytes/entry:  " << (double)total / data_.key_offsets.size() << "\n";
    }
};
```

## 总结

### 优化技术汇总

| 技术 | 适用场景 | 性能提升 |
|------|---------|---------|
| **数据布局优化** |
| Cache-Line对齐 | 多线程频繁访问 | 2-10x |
| AoS → SoA | SIMD处理 | 2-5x |
| Hot/Cold分离 | 部分数据频繁访问 | 1.5-3x |
| **压缩算法** |
| Snappy | 通用场景 | 2-3x空间，250MB/s |
| Zstd | 高压缩比需求 | 3-10x空间，100-400MB/s |
| LZ4 | 极速需求 | 1.5-2x空间，500MB/s |
| **编码技术** |
| VarInt | 小整数 | 50-87%空间节省 |
| Delta | 递增序列 | 3-10x空间节省 |
| 前缀压缩 | 相似key | 1.5-3x空间节省 |

### 最佳实践

1. **根据访问模式选择布局**
   - 随机访问 → Cache对齐
   - 顺序处理 → SoA + SIMD
   - 频繁更新 → Hot/Cold分离

2. **根据数据特征选择压缩**
   - 文本/JSON → Zstd + 字典
   - 二进制/随机 → Snappy/LZ4
   - 小对象 → 先编码后压缩

3. **编码优化原则**
   - 小整数 → VarInt
   - 时间序列 → Delta/DoubleDelta
   - 有序key → 前缀压缩
   - 重复值 → Dictionary编码

4. **权衡考虑**
   - CPU vs 内存：压缩级别选择
   - 写入 vs 读取：压缩 vs 无压缩
   - 延迟 vs 吞吐：批处理 vs 实时

记住：**测量驱动优化，没有银弹！** 🚀
