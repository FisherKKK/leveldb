# C++极致性能优化-07-模板元编程与编译期计算

## 课程概述

模板元编程（Template Metaprogramming, TMP）是C++中最强大的性能优化技术之一。通过将计算从运行时转移到编译时，我们可以实现零运行时开销的抽象。C++11/14/17/20的新特性（constexpr、if constexpr、concepts）使得编译期编程更加实用和高效。

**学习目标：**
- 掌握constexpr编译期计算
- 理解模板元编程技巧
- 学习SFINAE与type traits
- 掌握if constexpr条件编译
- 理解C++20 concepts
- 学习零开销抽象设计
- 掌握表达式模板优化

**典型性能提升：**
- 编译期计算：100%消除运行时开销
- 类型擦除：0开销多态（vs虚函数10-30%开销）
- 表达式模板：消除临时对象（3-10倍加速）
- Policy-based设计：0开销的灵活性

### 1.3 constexpr算法

```cpp
// 编译期排序
template<typename T, size_t N>
constexpr void bubble_sort(T (&arr)[N]) {
    for (size_t i = 0; i < N - 1; i++) {
        for (size_t j = 0; j < N - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                T temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

// 编译期使用
constexpr int unsorted[] = {5, 2, 8, 1, 9};
constexpr auto sorted = []() {
    int arr[] = {5, 2, 8, 1, 9};
    bubble_sort(arr);
    return arr;
}();
// sorted = {1, 2, 5, 8, 9}，编译期完成！

// 编译期二分查找
constexpr int binary_search(const int* arr, int n, int target) {
    int left = 0, right = n - 1;
    while (left <= right) {
        int mid = (left + right) / 2;
        if (arr[mid] == target) {
            return mid;
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    return -1;
}

constexpr int sorted_arr[] = {1, 3, 5, 7, 9, 11, 13};
constexpr int index = binary_search(sorted_arr, 7, 9);  // 4，编译期计算
```

```cpp
        if (*s1 != *s2) return false;
        s1++;
        s2++;
    }
    return *s1 == *s2;
}

// 编译期哈希
constexpr uint32_t fnv1a_hash(const char* str) {
    uint32_t hash = 2166136261u;
    while (*str) {
        hash ^= static_cast<uint32_t>(*str);
        hash *= 16777619u;
        str++;
    }
    return hash;
}

    switch (fnv1a_hash(cmd)) {
        case "start"_hash:
            start();
            break;

        case "pause"_hash:
```

### 2.3 编译期循环

#### 2.3.3 编译期for循环（C++17）

```cpp
// 模板递归实现编译期for
template<size_t I, size_t N, typename Func>
struct static_for_impl {
    static void call(Func&& func) {
        func(std::integral_constant<size_t, I>{});
        static_for_impl<I + 1, N, Func>::call(std::forward<Func>(func));
    }
};

template<size_t N, typename Func>
struct static_for_impl<N, N, Func> {
    static void call(Func&&) {}
};

template<size_t N, typename Func>
void static_for(Func&& func) {
    static_for_impl<0, N, Func>::call(std::forward<Func>(func));
}

// 使用：展开数组处理
void process_array() {
    int arr[5] = {1, 2, 3, 4, 5};

    static_for<5>([&](auto i) {
        // i是std::integral_constant<size_t, X>
        // 编译器为每个i生成独立代码
        arr[i] *= 2;
        // 展开为：arr[0] *= 2; arr[1] *= 2; ... arr[4] *= 2;
    });
}

// C++17 fold expression简化
template<size_t... Is, typename Func>
void static_for_fold(std::index_sequence<Is...>, Func&& func) {
    (func(std::integral_constant<size_t, Is>{}), ...);
}

template<size_t N, typename Func>
void static_for_modern(Func&& func) {
    static_for_fold(std::make_index_sequence<N>{}, std::forward<Func>(func));
}
```

#### 2.4.2 编译期Tuple

```cpp
// 简化的Tuple实现
template<typename... Ts>
class Tuple;

template<>
class Tuple<> {
public:
    static constexpr size_t size = 0;
};

template<typename T, typename... Ts>
class Tuple<T, Ts...> : private Tuple<Ts...> {
    T value_;

public:
    using Base = Tuple<Ts...>;
    static constexpr size_t size = 1 + Base::size;

    Tuple() = default;
    Tuple(T value, Ts... rest)
        : Base(rest...), value_(value) {}

    T& head() { return value_; }
    const T& head() const { return value_; }

    Base& tail() { return *this; }
    const Base& tail() const { return *this; }
};

// Get实现
template<size_t I, typename Tuple>
struct TupleElement;

template<typename T, typename... Ts>
struct TupleElement<0, Tuple<T, Ts...>> {
    using type = T;

    static T& get(Tuple<T, Ts...>& t) {
        return t.head();
    }
};

template<size_t I, typename T, typename... Ts>
struct TupleElement<I, Tuple<T, Ts...>> {
    using type = typename TupleElement<I - 1, Tuple<Ts...>>::type;

    static auto& get(Tuple<T, Ts...>& t) {
        return TupleElement<I - 1, Tuple<Ts...>>::get(t.tail());
    }
};

// 辅助函数
template<size_t I, typename... Ts>
auto& get(Tuple<Ts...>& t) {
    return TupleElement<I, Tuple<Ts...>>::get(t);
}

// 使用
Tuple<int, float, double> t(42, 3.14f, 2.718);
auto& first = get<0>(t);   // int& = 42
auto& second = get<1>(t);  // float& = 3.14
```

### 4.2 消除运行时分支

```cpp
// 运行时版本
template<typename T>
void serialize_runtime(std::ostream& os, const T& value) {
    if (std::is_same<T, int>::value) {
        os.write(reinterpret_cast<const char*>(&value), sizeof(int));
    } else if (std::is_same<T, float>::value) {
        os.write(reinterpret_cast<const char*>(&value), sizeof(float));
    } else {
        // fallback
    }
}
// 问题：运行时分支，即使类型已知

// 编译期版本
template<typename T>
void serialize_compiletime(std::ostream& os, const T& value) {
    if constexpr (std::is_same_v<T, int>) {
        os.write(reinterpret_cast<const char*>(&value), sizeof(int));
    } else if constexpr (std::is_same_v<T, float>) {
        os.write(reinterpret_cast<const char*>(&value), sizeof(float));
    } else {
        // 其他类型在编译时确定
        value.serialize(os);
    }
}
// 优势：编译时确定分支，生成特定代码，0运行时开销
```

## 第5章 C++20 Concepts

### 5.1 Concepts基础

```cpp
#include <concepts>

// 定义concept
template<typename T>
concept Numeric = std::is_arithmetic_v<T>;

template<typename T>
concept Addable = requires(T a, T b) {
    { a + b } -> std::convertible_to<T>;
};

template<typename T>
concept Container = requires(T c) {
    { c.begin() } -> std::input_or_output_iterator;
    { c.end() } -> std::input_or_output_iterator;
    { c.size() } -> std::convertible_to<size_t>;
};

// 使用concept约束函数
template<Numeric T>
T add(T a, T b) {
    return a + b;
}

// 或者
template<typename T>
    requires Numeric<T>
T subtract(T a, T b) {
    return a - b;
}

// 或者
template<typename T>
T multiply(T a, T b) requires Numeric<T> {
    return a * b;
}

// 使用
auto r1 = add(1, 2);         // OK
```

### 6.3 CRTP (Curiously Recurring Template Pattern)

```cpp
// 静态多态
template<typename Derived>
class Shape {
public:
    void draw() {
        static_cast<Derived*>(this)->draw_impl();  // 编译期绑定
    }

    double area() {
        return static_cast<Derived*>(this)->area_impl();
    }
};

class Circle : public Shape<Circle> {
    double radius_;

public:
    Circle(double r) : radius_(r) {}

    void draw_impl() {
        std::cout << "Drawing circle\n";
    }

    double area_impl() {
        return 3.14159 * radius_ * radius_;
    }
};

class Rectangle : public Shape<Rectangle> {
    double width_, height_;

public:
    Rectangle(double w, double h) : width_(w), height_(h) {}

    void draw_impl() {
        std::cout << "Drawing rectangle\n";
    }

    double area_impl() {
        return width_ * height_;
    }
};

// 使用

void process_shape(Shape<T>& shape) {

}

Circle circle(5.0);
Rectangle rect(4.0, 6.0);
process_shape(circle);     // 0开销
process_shape(rect);       // 0开销
```

## 第7章 表达式模板

### 7.1 延迟计算

```cpp
// 问题：向量运算产生临时对象
std::vector<double> a, b, c, d;
std::vector<double> result = a + b + c + d;
// 实际执行：
// temp1 = a + b;      // 临时对象1
// temp2 = temp1 + c;  // 临时对象2
// result = temp2 + d; // 临时对象3
// 3次内存分配，3次遍历

// 表达式模板：延迟计算
template<typename E>
class VecExpr {
public:
    double operator[](size_t i) const {
        return static_cast<const E&>(*this)[i];
    }

    size_t size() const {
        return static_cast<const E&>(*this).size();
    }
};

class Vec : public VecExpr<Vec> {
    std::vector<double> data_;

public:
    Vec(size_t n) : data_(n) {}

    double operator[](size_t i) const { return data_[i]; }
    double& operator[](size_t i) { return data_[i]; }
    size_t size() const { return data_.size(); }

    // 从表达式赋值
    template<typename E>
    Vec& operator=(const VecExpr<E>& expr) {
        const E& e = static_cast<const E&>(expr);
        for (size_t i = 0; i < size(); i++) {
            data_[i] = e[i];  // 只遍历一次！
        }
        return *this;
    }
};

// 加法表达式
template<typename E1, typename E2>
class VecAdd : public VecExpr<VecAdd<E1, E2>> {
    const E1& lhs_;
    const E2& rhs_;

public:
    VecAdd(const E1& lhs, const E2& rhs) : lhs_(lhs), rhs_(rhs) {}

    double operator[](size_t i) const {
        return lhs_[i] + rhs_[i];  // 延迟计算
    }

    size_t size() const { return lhs_.size(); }
};

// 运算符重载
template<typename E1, typename E2>
VecAdd<E1, E2> operator+(const VecExpr<E1>& lhs, const VecExpr<E2>& rhs) {
    return VecAdd<E1, E2>(static_cast<const E1&>(lhs),
                           static_cast<const E2&>(rhs));
}

// 使用
Vec a(1000), b(1000), c(1000), d(1000);
Vec result(1000);
result = a + b + c + d;
// 编译器生成：
// for (i = 0; i < 1000; i++) {
//     result[i] = a[i] + b[i] + c[i] + d[i];
// }
// 0临时对象，1次遍历！性能提升：3-10倍
```

### 8.3 案例3：高性能矩阵库

```cpp
// 使用表达式模板的矩阵库
template<typename E>
class MatExpr {
public:
    double operator()(size_t i, size_t j) const {
        return static_cast<const E&>(*this)(i, j);
    }

    size_t rows() const { return static_cast<const E&>(*this).rows(); }
    size_t cols() const { return static_cast<const E&>(*this).cols(); }
};

class Matrix : public MatExpr<Matrix> {
    std::vector<double> data_;
    size_t rows_, cols_;

public:
    Matrix(size_t r, size_t c) : rows_(r), cols_(c), data_(r * c) {}

    double operator()(size_t i, size_t j) const {
        return data_[i * cols_ + j];
    }

    double& operator()(size_t i, size_t j) {
        return data_[i * cols_ + j];
    }

    size_t rows() const { return rows_; }
    size_t cols() const { return cols_; }

    template<typename E>
    Matrix& operator=(const MatExpr<E>& expr) {
        const E& e = static_cast<const E&>(expr);
        for (size_t i = 0; i < rows_; i++) {
            for (size_t j = 0; j < cols_; j++) {
                (*this)(i, j) = e(i, j);  // 延迟计算
            }
        }
        return *this;
    }
};

// 矩阵加法表达式
template<typename E1, typename E2>
class MatAdd : public MatExpr<MatAdd<E1, E2>> {
    const E1& lhs_;
    const E2& rhs_;

public:
    MatAdd(const E1& lhs, const E2& rhs) : lhs_(lhs), rhs_(rhs) {}

    double operator()(size_t i, size_t j) const {
        return lhs_(i, j) + rhs_(i, j);
    }

    size_t rows() const { return lhs_.rows(); }
    size_t cols() const { return lhs_.cols(); }
};

// 矩阵乘法表达式（可优化为BLAS调用）
template<typename E1, typename E2>
class MatMul : public MatExpr<MatMul<E1, E2>> {
    const E1& lhs_;
    const E2& rhs_;

public:
    MatMul(const E1& lhs, const E2& rhs) : lhs_(lhs), rhs_(rhs) {}

    double operator()(size_t i, size_t j) const {
        double sum = 0;
        for (size_t k = 0; k < lhs_.cols(); k++) {
            sum += lhs_(i, k) * rhs_(k, j);
        }
        return sum;
    }

    size_t rows() const { return lhs_.rows(); }
    size_t cols() const { return rhs_.cols(); }
};

// 运算符
template<typename E1, typename E2>
MatAdd<E1, E2> operator+(const MatExpr<E1>& lhs, const MatExpr<E2>& rhs) {
    return MatAdd<E1, E2>(static_cast<const E1&>(lhs),
                           static_cast<const E2&>(rhs));
}

// 使用

Matrix result(100, 100);
result = A + B + C;  // 无临时对象，一次遍历
```
