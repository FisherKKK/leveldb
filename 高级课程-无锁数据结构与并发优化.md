## 1. C++内存模型

### 1.1 内存模型基础

```cpp
#include <atomic>
#include <thread>
#include <iostream>

// C++内存模型的核心概念：
// 1. 原子性 (Atomicity)
// 2. 可见性 (Visibility)
// 3. 顺序性 (Ordering)

// 问题示例：数据竞争
int x = 0;  // 非原子变量
int y = 0;

void thread1() {
    x = 1;  // 写操作1
    y = 1;  // 写操作2
}

void thread2() {
    while (y == 0);  // 等待y变化
    std::cout << x << std::endl;  // 可能输出0！
}

// 为什么可能输出0？
// 1. 编译器可能重排指令 (x和y的写顺序)
// 2. CPU可能乱序执行
// 3. Cache一致性延迟

// 解决方案：原子变量
std::atomic<int> atomic_x{0};
std::atomic<int> atomic_y{0};

void thread1_fixed() {
    atomic_x.store(1, std::memory_order_release);  // 释放语义
    atomic_y.store(1, std::memory_order_release);
}

void thread2_fixed() {
    while (atomic_y.load(std::memory_order_acquire) == 0);  // 获取语义
    std::cout << atomic_x.load(std::memory_order_acquire) << std::endl;  // 保证是1
}
```

### 1.2 happens-before关系

```cpp
// happens-before: 定义操作间的偏序关系

// 规则1：单线程内，前面的操作happens-before后面的操作
void single_thread() {
    int a = 1;  // (1)
    int b = 2;  // (2) happens-before (1)
    int c = a + b;  // (3) happens-before (1) and (2)
}

// 规则2：解锁happens-before后续的加锁
std::mutex m;
int shared = 0;

void thread_a() {
    std::lock_guard<std::mutex> lock(m);
    shared = 42;  // (1)
}  // unlock happens-before (2)

void thread_b() {
    std::lock_guard<std::mutex> lock(m);  // (2)
    std::cout << shared << std::endl;  // 保证看到42
}

// 规则3：原子操作的release-acquire
std::atomic<int> flag{0};
int data = 0;

void producer() {
    data = 100;  // (1)
    flag.store(1, std::memory_order_release);  // (2) release
}

void consumer() {
    while (flag.load(std::memory_order_acquire) == 0);  // (3) acquire
    std::cout << data << std::endl;  // (4) 保证看到100
}
// (1) happens-before (2) happens-before (3) happens-before (4)
```

## 2. 原子操作深度剖析

### 2.1 原子操作类型

```cpp
#include <atomic>

// 原子类型
std::atomic<int> atomic_int;
std::atomic<bool> atomic_bool;
std::atomic<void*> atomic_ptr;

// 常用原子操作
void atomic_operations() {
    std::atomic<int> x{0};

    // 1. Load/Store
    int val = x.load(std::memory_order_acquire);
    x.store(42, std::memory_order_release);

    // 2. Exchange（原子交换）
    int old = x.exchange(100);  // 返回旧值，设置新值

    // 3. Compare-And-Swap (CAS)
    int expected = 100;
    bool success = x.compare_exchange_strong(expected, 200);
    // 如果x==expected，则x=200，返回true
    // 否则，expected=x，返回false

    // 4. Fetch-And-Add
    int prev = x.fetch_add(10);  // 原子加法，返回旧值

    // 5. Fetch-And-Or/And/Xor
    x.fetch_or(0xFF);
    x.fetch_and(0x0F);
    x.fetch_xor(0x55);
}
```

### 2.2 CAS循环模式

```cpp
// CAS循环：无锁算法的基础模式
template<typename T>
void atomic_increment(std::atomic<T>& value) {
    T old_val = value.load(std::memory_order_relaxed);
    while (!value.compare_exchange_weak(
        old_val,                          // 期望值
        old_val + 1,                      // 新值
        std::memory_order_release,        // 成功时的内存顺序
        std::memory_order_relaxed)) {     // 失败时的内存顺序
        // CAS失败，old_val已更新为当前值，重试
    }
}

// compare_exchange_weak vs strong
// - weak: 可能伪失败 (spurious failure)，但更快
// - strong: 不会伪失败，但可能稍慢
// 规则：在循环中用weak，单次操作用strong

// 实战示例：无锁计数器
class LockFreeCounter {
private:
    std::atomic<uint64_t> count_{0};

public:
    uint64_t increment() {
        uint64_t old_val = count_.load(std::memory_order_relaxed);
        while (!count_.compare_exchange_weak(
            old_val, old_val + 1,
            std::memory_order_acq_rel,
            std::memory_order_relaxed));
        return old_val + 1;
    }

    uint64_t get() const {
        return count_.load(std::memory_order_acquire);
    }
};

// 性能测试
void benchmark_counter() {
    const int num_threads = 8;
    const int iterations = 1000000;

    // 测试1：互斥锁
    {
        uint64_t counter = 0;
        std::mutex mutex;

        auto start = std::chrono::high_resolution_clock::now();

        std::vector<std::thread> threads;
        for (int i = 0; i < num_threads; i++) {
            threads.emplace_back([&]() {
                for (int j = 0; j < iterations; j++) {
                    std::lock_guard<std::mutex> lock(mutex);
                    counter++;
                }
            });
        }

        for (auto& t : threads) t.join();

        auto mutex_time = std::chrono::high_resolution_clock::now() - start;
        std::cout << "Mutex:        " << mutex_time.count() / 1e6 << " ms\n";
    }

    // 测试2：无锁计数器
    {
        LockFreeCounter counter;

        auto start = std::chrono::high_resolution_clock::now();

        std::vector<std::thread> threads;
        for (int i = 0; i < num_threads; i++) {
            threads.emplace_back([&]() {
                for (int j = 0; j < iterations; j++) {
                    counter.increment();
                }
            });
        }

        for (auto& t : threads) t.join();

        auto lockfree_time = std::chrono::high_resolution_clock::now() - start;
        std::cout << "Lock-Free:    " << lockfree_time.count() / 1e6 << " ms\n";
    }
}

// 典型输出：
// Mutex:        3850 ms
// Lock-Free:    720 ms (5.3x faster)
```

## 3. 内存顺序详解

### 3.1 六种内存顺序

```cpp
// C++11定义的六种内存顺序：
enum memory_order {
    memory_order_relaxed,  // 最弱：无同步，仅保证原子性
    memory_order_consume,  // 数据依赖顺序（很少用）
    memory_order_acquire,  // 获取：后续操作不能重排到前面
    memory_order_release,  // 释放：前面操作不能重排到后面
    memory_order_acq_rel,  // 获取+释放
    memory_order_seq_cst   // 顺序一致：最强，全局顺序
};

// 1. memory_order_relaxed
// 只保证原子性，不保证顺序
std::atomic<int> x{0}, y{0};

void relaxed_example() {
    // 线程1
    x.store(1, std::memory_order_relaxed);
    y.store(1, std::memory_order_relaxed);

    // 线程2可能看到：
    // x=0, y=1  或 x=1, y=0  或 x=1, y=1
}

// 2. memory_order_acquire / release
// 用于同步：release写入，acquire读取
std::atomic<bool> ready{false};
int data = 0;

void producer_release() {
    data = 42;  // (1)
    ready.store(true, std::memory_order_release);  // (2)
    // (1) happens-before (2)
}

void consumer_acquire() {
    while (!ready.load(std::memory_order_acquire));  // (3)
    assert(data == 42);  // (4)
    // (3) happens-before (4)
}

// 3. memory_order_seq_cst (默认)
// 顺序一致性：最强保证，但最慢
std::atomic<int> seq_x{0}, seq_y{0};

void thread_seq_1() {
    seq_x.store(1);  // 默认seq_cst
}

void thread_seq_2() {
    seq_y.store(1);
}

void thread_seq_3() {
    while (seq_x.load() == 0);
    assert(seq_y.load() == 1);  // 可能失败！
}

void thread_seq_4() {
    while (seq_y.load() == 0);
    assert(seq_x.load() == 1);  // 可能失败！
}
// seq_cst保证所有线程看到相同的修改顺序
```

## 5. 无锁栈

### 5.1 Treiber Stack

```cpp
// Treiber Stack: 经典无锁栈
template<typename T>
class LockFreeStack {
private:
    struct Node {
        T data;
        Node* next;

        Node(const T& val) : data(val), next(nullptr) {}
    };

    std::atomic<Node*> head_{nullptr};

public:
    ~LockFreeStack() {
        while (pop());
    }

    void push(const T& value) {
        Node* new_node = new Node(value);
        new_node->next = head_.load(std::memory_order_relaxed);

        // CAS循环：尝试设置head
        while (!head_.compare_exchange_weak(
            new_node->next,                // expected = current head
            new_node,                      // new head
            std::memory_order_release,     // success
            std::memory_order_relaxed)) {  // failure
            // CAS失败，重试
        }
    }

    bool pop(T& result) {
        Node* old_head = head_.load(std::memory_order_relaxed);

        while (old_head &&
               !head_.compare_exchange_weak(
                   old_head,                   // expected
                   old_head->next,             // new head
                   std::memory_order_acquire,  // success
                   std::memory_order_relaxed)) {  // failure
            // CAS失败，重试
        }

        if (old_head) {
            result = old_head->data;
            delete old_head;  // ⚠️ ABA问题！
            return true;
        }

        return false;
    }

    bool pop() {
        T dummy;
        return pop(dummy);
    }
};

// 性能测试
void benchmark_stack() {
    const int num_threads = 8;
    const int operations = 1000000;

    // 测试1：std::stack + mutex
    {
        std::stack<int> stack;
        std::mutex mutex;

        auto start = std::chrono::high_resolution_clock::now();

        std::vector<std::thread> threads;
        for (int i = 0; i < num_threads; i++) {
            threads.emplace_back([&]() {
                for (int j = 0; j < operations; j++) {
                    {
                        std::lock_guard<std::mutex> lock(mutex);
                        stack.push(j);
                    }
                    {
                        std::lock_guard<std::mutex> lock(mutex);
                        if (!stack.empty()) stack.pop();
                    }
                }
            });
        }

        for (auto& t : threads) t.join();

        auto mutex_time = std::chrono::high_resolution_clock::now() - start;
        std::cout << "Mutex Stack:      " << mutex_time.count() / 1e6 << " ms\n";
    }

    // 测试2：Lock-Free Stack
    {
        LockFreeStack<int> stack;

        auto start = std::chrono::high_resolution_clock::now();

        std::vector<std::thread> threads;
        for (int i = 0; i < num_threads; i++) {
            threads.emplace_back([&]() {
                for (int j = 0; j < operations; j++) {
                    stack.push(j);
                    int val;
                    stack.pop(val);
                }
            });
        }

        for (auto& t : threads) t.join();

        auto lockfree_time = std::chrono::high_resolution_clock::now() - start;
        std::cout << "Lock-Free Stack:  " << lockfree_time.count() / 1e6 << " ms\n";
        std::cout << "Speedup:          "
                  << (double)mutex_time.count() / lockfree_time.count() << "x\n";
    }
}

// 典型输出：
// Mutex Stack:      5200 ms
// Lock-Free Stack:  980 ms
// Speedup:          5.3x
```

## 6. 无锁队列

### 6.1 Michael-Scott队列

```cpp
// Michael-Scott Queue: 最著名的无锁队列
template<typename T>
class LockFreeQueue {
private:
    struct Node {
        std::shared_ptr<T> data;
        std::atomic<Node*> next;

        Node() : next(nullptr) {}
    };

    std::atomic<Node*> head_;
    std::atomic<Node*> tail_;

public:
    LockFreeQueue() {
        Node* dummy = new Node();
        head_.store(dummy);
        tail_.store(dummy);
    }

    ~LockFreeQueue() {
        while (pop());
        delete head_.load();
    }

    void enqueue(const T& value) {
        auto new_data = std::make_shared<T>(value);
        Node* new_node = new Node();
        new_node->data = new_data;

        Node* old_tail = tail_.load();

        while (true) {
            Node* tail_next = old_tail->next.load();

            // 检查tail是否真的是末尾
            if (tail_next == nullptr) {
                // 尝试链接新节点
                if (old_tail->next.compare_exchange_weak(tail_next, new_node)) {
                    // 成功！尝试更新tail
                    tail_.compare_exchange_weak(old_tail, new_node);
                    return;
                }
            } else {
                // tail落后了，帮助推进
                tail_.compare_exchange_weak(old_tail, tail_next);
            }

            old_tail = tail_.load();
        }
    }

    bool dequeue(T& result) {
        while (true) {
            Node* old_head = head_.load();
            Node* old_tail = tail_.load();
            Node* head_next = old_head->next.load();

            // 检查head是否一致
            if (old_head == head_.load()) {
                // 队列为空？
                if (old_head == old_tail) {
                    if (head_next == nullptr) {
                        return false;  // 真的空
                    }
                    // tail落后，帮助推进
                    tail_.compare_exchange_weak(old_tail, head_next);
                } else {
                    // 读取数据
                    result = *head_next->data;

                    // 尝试推进head
                    if (head_.compare_exchange_weak(old_head, head_next)) {
                        delete old_head;
                        return true;
                    }
                }
            }
        }
    }

    bool pop() {
        T dummy;
        return dequeue(dummy);
    }
};
```

### 6.2 单生产者单消费者队列（极速）

```cpp
// SPSC Queue: Single Producer Single Consumer
// 无需CAS，只需要memory_order控制
template<typename T>
class SPSCQueue {
private:
    struct Node {
        T data;
        std::atomic<Node*> next;

        Node() : next(nullptr) {}
        Node(const T& val) : data(val), next(nullptr) {}
    };

    alignas(64) std::atomic<Node*> head_;  // 消费者访问
    alignas(64) std::atomic<Node*> tail_;  // 生产者访问
    // 对齐到不同cache line，避免false sharing

public:
    SPSCQueue() {
        Node* dummy = new Node();
        head_.store(dummy, std::memory_order_relaxed);
        tail_.store(dummy, std::memory_order_relaxed);
    }

    ~SPSCQueue() {
        while (pop());
        delete head_.load();
    }

    // 生产者：只有一个线程调用
    void enqueue(const T& value) {
        Node* new_node = new Node(value);
        Node* prev_tail = tail_.load(std::memory_order_relaxed);

        prev_tail->next.store(new_node, std::memory_order_release);
        tail_.store(new_node, std::memory_order_relaxed);
    }

    // 消费者：只有一个线程调用
    bool dequeue(T& result) {
        Node* old_head = head_.load(std::memory_order_relaxed);
        Node* next = old_head->next.load(std::memory_order_acquire);

        if (next == nullptr) {
            return false;  // 队列空
        }

        result = next->data;
        head_.store(next, std::memory_order_relaxed);
        delete old_head;

        return true;
    }

    bool pop() {
        T dummy;
        return dequeue(dummy);
    }
};

// SPSC vs MPMC性能对比
void benchmark_queues() {
    const int messages = 10000000;

    std::cout << "=== Queue Performance ===\n\n";

    // 测试1: SPSC Queue
    {
        SPSCQueue<int> queue;

        auto start = std::chrono::high_resolution_clock::now();

        std::thread producer([&]() {
            for (int i = 0; i < messages; i++) {
                queue.enqueue(i);
            }
        });

        std::thread consumer([&]() {
            int val;
            for (int i = 0; i < messages; i++) {
                while (!queue.dequeue(val));
            }
        });

        producer.join();
        consumer.join();

        auto spsc_time = std::chrono::high_resolution_clock::now() - start;
        double spsc_throughput = messages / (spsc_time.count() / 1e9) / 1e6;

        std::cout << "SPSC Queue:  " << spsc_time.count() / 1e6 << " ms, "
                  << spsc_throughput << " Mops/s\n";
    }

    // 测试2: MPMC Queue
    {
        LockFreeQueue<int> queue;

        auto start = std::chrono::high_resolution_clock::now();

        std::thread producer([&]() {
            for (int i = 0; i < messages; i++) {
                queue.enqueue(i);
            }
        });

        std::thread consumer([&]() {
            int val;
            for (int i = 0; i < messages; i++) {
                while (!queue.dequeue(val));
            }
        });

        producer.join();
        consumer.join();

        auto mpmc_time = std::chrono::high_resolution_clock::now() - start;
        double mpmc_throughput = messages / (mpmc_time.count() / 1e9) / 1e6;

        std::cout << "MPMC Queue:  " << mpmc_time.count() / 1e6 << " ms, "
                  << mpmc_throughput << " Mops/s\n";
    }

    std::cout << "\n典型结果：\n";
    std::cout << "- SPSC: 30-50 Mops/s (无CAS，极快)\n";
    std::cout << "- MPMC: 5-15 Mops/s (需要CAS)\n";
}
```

## 8. LevelDB SkipList

### 8.1 LevelDB中的无锁SkipList

```cpp
// LevelDB的SkipList：无锁读取，单写入者
// 特点：
// 1. 多线程并发读
// 2. 单线程写（由外部保证）
// 3. 读写可以并发
// 4. 无需内存回收（表生命周期内节点不删除）

template<typename Key, class Comparator>
class SkipList {
private:
    struct Node {
        Key const key;

        // 可变长度的next数组
        std::atomic<Node*> next_[1];

        // 获取指定层的next节点
        Node* Next(int n) {
            return next_[n].load(std::memory_order_acquire);
        }

        void SetNext(int n, Node* x) {
            next_[n].store(x, std::memory_order_release);
        }

        // No-barrier版本：仅在新节点尚未发布时使用
        Node* NoBarrier_Next(int n) {
            return next_[n].load(std::memory_order_relaxed);
        }

        void NoBarrier_SetNext(int n, Node* x) {
            next_[n].store(x, std::memory_order_relaxed);
        }
    };

    enum { kMaxHeight = 12 };

    Comparator const compare_;
    Arena* const arena_;  // 内存分配器

    Node* const head_;
    std::atomic<int> max_height_;  // 当前最大高度

    Random rnd_;  // 用于随机高度

public:
    explicit SkipList(Comparator cmp, Arena* arena)
        : compare_(cmp),
          arena_(arena),
          head_(NewNode(0 /* any key */, kMaxHeight)),
          max_height_(1),
          rnd_(0xdeadbeef) {
        for (int i = 0; i < kMaxHeight; i++) {
            head_->SetNext(i, nullptr);
        }
    }

    // 插入（单线程写入）
    void Insert(const Key& key) {
        Node* prev[kMaxHeight];
        Node* x = FindGreaterOrEqual(key, prev);

        // 随机高度
        int height = RandomHeight();
        if (height > GetMaxHeight()) {
            for (int i = GetMaxHeight(); i < height; i++) {
                prev[i] = head_;
            }
            max_height_.store(height, std::memory_order_relaxed);
        }

        x = NewNode(key, height);

        for (int i = 0; i < height; i++) {
            // NoBarrier: x尚未发布
            x->NoBarrier_SetNext(i, prev[i]->NoBarrier_Next(i));

            // 使用release语义发布节点
            prev[i]->SetNext(i, x);
        }
    }

    // 查找（多线程读取）
    bool Contains(const Key& key) const {
        Node* x = FindGreaterOrEqual(key, nullptr);
        if (x != nullptr && Equal(key, x->key)) {
            return true;
        } else {
            return false;
        }
    }

private:
    Node* NewNode(const Key& key, int height) {
        char* mem = arena_->AllocateAligned(
            sizeof(Node) + sizeof(std::atomic<Node*>) * (height - 1));
        return new (mem) Node(key);
    }

    int RandomHeight() {
        static const unsigned int kBranching = 4;
        int height = 1;
        while (height < kMaxHeight && ((rnd_.Next() % kBranching) == 0)) {
            height++;
        }
        return height;
    }

    int GetMaxHeight() const {
        return max_height_.load(std::memory_order_relaxed);
    }

    bool Equal(const Key& a, const Key& b) const {
        return (compare_(a, b) == 0);
    }

    Node* FindGreaterOrEqual(const Key& key, Node** prev) const {
        Node* x = head_;
        int level = GetMaxHeight() - 1;

        while (true) {
            Node* next = x->Next(level);  // acquire语义

            if (KeyIsAfterNode(key, next)) {
                x = next;
            } else {
                if (prev != nullptr) prev[level] = x;
                if (level == 0) {
                    return next;
                } else {
                    level--;
                }
            }
        }
    }

    bool KeyIsAfterNode(const Key& key, Node* n) const {
        return (n != nullptr) && (compare_(n->key, key) < 0);
    }
};

// 性能测试
void benchmark_skiplist() {
    Arena arena;
    SkipList<uint64_t, std::less<uint64_t>> skiplist(std::less<uint64_t>(), &arena);

    const int num_threads = 8;
```

```cpp
    T* ptr;

    TaggedPointer(T* p, uintptr_t t) : ptr(p), tag(t) {}

class LockFreeStackNoABA {
private:
    struct Node {
        T data;
        Node* next;
    };

    struct TaggedPointer {
        Node* ptr;
        uint64_t tag;
    };

    std::atomic<__int128> head_;  // 128位原子变量

public:
    void push(const T& value) {
        Node* new_node = new Node{value, nullptr};

        __int128 old_head = head_.load();

        do {
            TaggedPointer* old = reinterpret_cast<TaggedPointer*>(&old_head);
            new_node->next = old->ptr;

            __int128 new_head;
            TaggedPointer* new_tagged = reinterpret_cast<TaggedPointer*>(&new_head);
            new_tagged->ptr = new_node;
            new_tagged->tag = old->tag + 1;  // 递增版本号

        } while (!head_.compare_exchange_weak(old_head, new_head));
    }

    bool pop(T& result) {
        __int128 old_head = head_.load();

        while (true) {
            TaggedPointer* old = reinterpret_cast<TaggedPointer*>(&old_head);

            if (old->ptr == nullptr) {
                return false;
            }

            __int128 new_head;
            TaggedPointer* new_tagged = reinterpret_cast<TaggedPointer*>(&new_head);
            new_tagged->ptr = old->ptr->next;
            new_tagged->tag = old->tag + 1;  // 递增版本号

            if (head_.compare_exchange_weak(old_head, new_head)) {
                result = old->ptr->data;
                delete old->ptr;
                return true;
            }
        }
    }
};
```

## 10. Hazard Pointer

### 10.1 Hazard Pointer实现

```cpp
// Hazard Pointer: 无GC语言的安全内存回收
// 原理：
// 1. 每个线程有若干hazard pointer槽位
// 2. 访问节点前，先将指针写入槽位
// 3. 删除节点时，检查是否有hazard pointer指向它
// 4. 如果有，延迟删除

template<typename T>
class HazardPointerManager {
private:
    static constexpr int kMaxThreads = 128;
    static constexpr int kHazardPtrPerThread = 2;

    struct HazardPointer {
        std::atomic<T*> ptr{nullptr};
        std::atomic<std::thread::id> owner{std::thread::id()};
    };

    struct RetiredNode {
        T* ptr;
        std::function<void(T*)> deleter;
    };

    alignas(64) HazardPointer hazard_ptrs_[kMaxThreads * kHazardPtrPerThread];
    std::atomic<int> hp_count_{0};

    // 每个线程的退休列表
    thread_local static std::vector<RetiredNode> retired_list_;
    thread_local static HazardPointer* my_hazards_[kHazardPtrPerThread];

public:
    // 获取hazard pointer
    HazardPointer* acquire() {
        std::thread::id tid = std::this_thread::get_id();

        // 查找空闲槽位
        for (int i = 0; i < kMaxThreads * kHazardPtrPerThread; i++) {
            std::thread::id empty_id;
            if (hazard_ptrs_[i].owner.compare_exchange_strong(
                empty_id, tid,
                std::memory_order_acquire,
                std::memory_order_relaxed)) {
                return &hazard_ptrs_[i];
            }
        }

        return nullptr;  // 无可用槽位
    }

    // 释放hazard pointer
    void release(HazardPointer* hp) {
        hp->ptr.store(nullptr, std::memory_order_release);
        hp->owner.store(std::thread::id(), std::memory_order_release);
    }

    // 保护指针
    template<typename AtomicPtr>
    T* protect(AtomicPtr& atomic_ptr, HazardPointer* hp) {
        T* ptr = nullptr;
        T* old_ptr = atomic_ptr.load(std::memory_order_relaxed);

        do {
            ptr = old_ptr;
            hp->ptr.store(ptr, std::memory_order_release);
            old_ptr = atomic_ptr.load(std::memory_order_acquire);
        } while (ptr != old_ptr);  // 确保一致性

        return ptr;
    }

    // 延迟删除
    void retire(T* ptr, std::function<void(T*)> deleter) {
        retired_list_.push_back({ptr, deleter});

        // 定期扫描
        if (retired_list_.size() >= 100) {
            scan();
        }
    }

private:
    void scan() {
        // 收集所有hazard pointers
        std::unordered_set<T*> hazard_set;
        for (int i = 0; i < kMaxThreads * kHazardPtrPerThread; i++) {
            T* ptr = hazard_ptrs_[i].ptr.load(std::memory_order_acquire);
            if (ptr) {
                hazard_set.insert(ptr);
            }
        }

        // 删除不在hazard set中的节点
        auto it = retired_list_.begin();
        while (it != retired_list_.end()) {
            if (hazard_set.find(it->ptr) == hazard_set.end()) {
                // 安全删除
                it->deleter(it->ptr);
                it = retired_list_.erase(it);
            } else {
                ++it;
            }
        }
    }
};

// 使用Hazard Pointer的无锁栈
template<typename T>
class SafeLockFreeStack {
private:
    struct Node {
        T data;
        Node* next;
    };

    std::atomic<Node*> head_{nullptr};
    HazardPointerManager<Node> hp_manager_;

public:
    void push(const T& value) {
        Node* new_node = new Node{value, head_.load()};

        while (!head_.compare_exchange_weak(
            new_node->next, new_node,
            std::memory_order_release,
            std::memory_order_relaxed));
    }

    bool pop(T& result) {
        HazardPointer* hp = hp_manager_.acquire();
        if (!hp) return false;

        while (true) {
            Node* old_head = hp_manager_.protect(head_, hp);

            if (!old_head) {
                hp_manager_.release(hp);
                return false;
            }

            if (head_.compare_exchange_weak(
                old_head, old_head->next,
                std::memory_order_acquire,
                std::memory_order_relaxed)) {

                result = old_head->data;

                hp_manager_.release(hp);

                // 延迟删除
                hp_manager_.retire(old_head, [](Node* n) { delete n; });

                return true;
            }
        }
    }
};
```

```cpp
// 综合应用：SPSC队列 + 无锁写入

class LockFreeLogger {
private:
    struct LogEntry {
        uint64_t timestamp;
        int level;
        std::string message;
    };

    SPSCQueue<LogEntry> queue_;
    std::atomic<bool> running_{true};
    std::thread writer_thread_;
    int fd_;

public:
    LockFreeLogger(const char* filename) {
        fd_ = open(filename, O_WRONLY | O_CREAT | O_APPEND, 0644);

        // 启动写入线程
        writer_thread_ = std::thread([this]() {
            write_loop();
        });
    }

    ~LockFreeLogger() {
        running_.store(false);
        writer_thread_.join();
        close(fd_);
    }

    // 生产者：应用线程调用
    void log(int level, const std::string& message) {
        queue_.enqueue({
            .timestamp = now_microseconds(),
            .level = level,
            .message = message
        });
    }

private:
    // 消费者：单独写入线程
    void write_loop() {
        std::vector<LogEntry> batch;
        batch.reserve(1000);

        while (running_.load(std::memory_order_relaxed)) {
            // 批量读取
            LogEntry entry;
            while (queue_.dequeue(entry)) {
                batch.push_back(std::move(entry));

                if (batch.size() >= 1000) break;
            }

            if (!batch.empty()) {
                // 批量写入
                for (const auto& e : batch) {
                    char buffer[4096];
                    int len = format_log(buffer, sizeof(buffer), e);
                    write(fd_, buffer, len);
                }

                batch.clear();
            }

            // 避免busy-wait
            if (batch.empty()) {
                std::this_thread::sleep_for(std::chrono::microseconds(100));
            }
        }
    }

    int format_log(char* buffer, size_t size, const LogEntry& entry) {
        return snprintf(buffer, size, "[%lu] [%d] %s\n",
                       entry.timestamp, entry.level, entry.message.c_str());
    }

    uint64_t now_microseconds() {
        auto now = std::chrono::steady_clock::now();
        return std::chrono::duration_cast<std::chrono::microseconds>(
            now.time_since_epoch()).count();
    }
};

            }

    for (auto& t : threads) t.join();

    auto time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "Lock-Free Logger:\n";

    std::cout << "  Throughput: " << throughput / 1e6 << " M logs/s\n";
    std::cout << "  Latency:    " << (time.count() / (num_threads * logs_per_thread)) << " ns/log\n";
```
