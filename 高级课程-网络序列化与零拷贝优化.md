### 1.2 手写二进制协议（基准）

```cpp
// 最简单的二进制协议
class BinarySerializer {
public:
    // 序列化
    static std::vector<uint8_t> serialize(const Person& p) {
        std::vector<uint8_t> buffer;

        // ID (8 bytes)
        append_uint64(buffer, p.id);

        // Name (4 bytes length + data)
        append_string(buffer, p.name);

        // Email
        append_string(buffer, p.email);

        // Age (4 bytes)
        append_int32(buffer, p.age);

        // Tags count + tags
        append_uint32(buffer, p.tags.size());
        for (const auto& tag : p.tags) {
            append_string(buffer, tag);
        }

        return buffer;
    }

    // 反序列化
    static Person deserialize(const uint8_t* data, size_t size) {
        Person p;
        size_t offset = 0;

        p.id = read_uint64(data, offset);
        p.name = read_string(data, offset);
        p.email = read_string(data, offset);
        p.age = read_int32(data, offset);

        uint32_t tag_count = read_uint32(data, offset);
        for (uint32_t i = 0; i < tag_count; i++) {
            p.tags.push_back(read_string(data, offset));
        }

        return p;
    }

private:
    static void append_uint64(std::vector<uint8_t>& buf, uint64_t val) {
        for (int i = 0; i < 8; i++) {
            buf.push_back((val >> (i * 8)) & 0xFF);
        }
    }

    static void append_int32(std::vector<uint8_t>& buf, int32_t val) {
        for (int i = 0; i < 4; i++) {
            buf.push_back((val >> (i * 8)) & 0xFF);
        }
    }

    static void append_uint32(std::vector<uint8_t>& buf, uint32_t val) {
        for (int i = 0; i < 4; i++) {
            buf.push_back((val >> (i * 8)) & 0xFF);
        }
    }

    static void append_string(std::vector<uint8_t>& buf, const std::string& str) {
        append_uint32(buf, str.size());
        buf.insert(buf.end(), str.begin(), str.end());
    }

    static uint64_t read_uint64(const uint8_t* data, size_t& offset) {
        uint64_t val = 0;
        for (int i = 0; i < 8; i++) {
            val |= uint64_t(data[offset++]) << (i * 8);
        }
        return val;
    }

    static int32_t read_int32(const uint8_t* data, size_t& offset) {
        int32_t val = 0;
        for (int i = 0; i < 4; i++) {
            val |= int32_t(data[offset++]) << (i * 8);
        }
        return val;
    }

    static uint32_t read_uint32(const uint8_t* data, size_t& offset) {
        uint32_t val = 0;
        for (int i = 0; i < 4; i++) {
            val |= uint32_t(data[offset++]) << (i * 8);
        }
        return val;
    }

    static std::string read_string(const uint8_t* data, size_t& offset) {
        uint32_t len = read_uint32(data, offset);
        std::string str(reinterpret_cast<const char*>(data + offset), len);
        offset += len;
        return str;
    }
};

void benchmark_binary() {
    auto data = generate_test_data(10000);

    // 序列化
    auto start = std::chrono::high_resolution_clock::now();
    std::vector<std::vector<uint8_t>> serialized;
    for (const auto& person : data) {
        serialized.push_back(BinarySerializer::serialize(person));
    }
    auto serialize_time = std::chrono::high_resolution_clock::now() - start;

    // 反序列化
    start = std::chrono::high_resolution_clock::now();
    std::vector<Person> deserialized;
    for (const auto& buf : serialized) {
        deserialized.push_back(BinarySerializer::deserialize(buf.data(), buf.size()));
    }
    auto deserialize_time = std::chrono::high_resolution_clock::now() - start;

    // 计算大小
    size_t total_size = 0;
    for (const auto& buf : serialized) {
        total_size += buf.size();
    }

    std::cout << "Binary Protocol:\n";
    std::cout << "  Serialize:   " << serialize_time.count() / 1e6 << " ms\n";
    std::cout << "  Deserialize: " << deserialize_time.count() / 1e6 << " ms\n";
    std::cout << "  Total size:  " << total_size << " bytes\n";
    std::cout << "  Avg size:    " << total_size / data.size() << " bytes/record\n";
}
```

### 2.2 Protobuf优化技巧

```cpp
#include "person.pb.h"
#include <google/protobuf/io/coded_stream.h>
#include <google/protobuf/io/zero_copy_stream_impl.h>

// 技巧1：Arena分配器（减少malloc）
void protobuf_arena_example() {
    google::protobuf::Arena arena;

    // 从arena分配
    Person* person = google::protobuf::Arena::CreateMessage<Person>(&arena);
    person->set_id(1);
    person->set_name("Alice");
    person->set_email("alice@example.com");
    person->set_age(25);

    // 无需手动delete，arena析构时自动清理
}

// 技巧2：预分配字符串空间
void protobuf_preallocate() {
    Person person;
    person.set_id(1);

    // 预分配避免多次realloc
    std::string* name = person.mutable_name();
    name->reserve(100);
    *name = "Very long name...";

    // 或直接移动
    std::string email = "alice@example.com";
    person.set_allocated_email(&email);  // 零拷贝
}

// 技巧3：序列化到预分配buffer
void protobuf_serialize_to_buffer() {
    Person person;
    person.set_id(1);
    person.set_name("Alice");

    // 计算大小
    size_t size = person.ByteSizeLong();

    // 预分配buffer
    std::vector<uint8_t> buffer(size);

    // 直接序列化到buffer（避免中间string）
    google::protobuf::io::ArrayOutputStream aos(buffer.data(), buffer.size());
    google::protobuf::io::CodedOutputStream cos(&aos);
    person.SerializeToCodedStream(&cos);
}

// 技巧4：批量序列化（减少syscall）
void protobuf_batch_serialize(const std::vector<Person>& persons) {
    PersonList list;

    // 预分配
    list.mutable_persons()->Reserve(persons.size());

    // 批量添加
    for (const auto& p : persons) {
        *list.add_persons() = p;
    }

    // 一次性序列化
    std::string output;
    list.SerializeToString(&output);
}
```

### 3.2 FlatBuffers零拷贝特性

```cpp
#include "person_generated.h"
#include <flatbuffers/flatbuffers.h>

using namespace MyGame;

// FlatBuffers的核心优势：
// 1. 序列化后可以直接访问，无需解析
// 2. 内存布局对齐，直接指针访问
// 3. 向前/向后兼容

void flatbuffers_example() {
    flatbuffers::FlatBufferBuilder builder(1024);

    // 构建字符串（必须先构建，因为是偏移量）
    auto name = builder.CreateString("Alice");
    auto email = builder.CreateString("alice@example.com");

    // 构建tags vector
    std::vector<flatbuffers::Offset<flatbuffers::String>> tags_vector;
    tags_vector.push_back(builder.CreateString("tag1"));
    tags_vector.push_back(builder.CreateString("tag2"));
    auto tags = builder.CreateVector(tags_vector);

    // 构建Person
    auto person = CreatePerson(builder, 1, name, email, 25, tags);

    builder.Finish(person);

    // 获取buffer（零拷贝）
    uint8_t* buf = builder.GetBufferPointer();
    size_t size = builder.GetSize();

    // 直接访问（无需解析！）
    auto parsed = GetPerson(buf);
    std::cout << "ID: " << parsed->id() << "\n";
    std::cout << "Name: " << parsed->name()->c_str() << "\n";
    std::cout << "Age: " << parsed->age() << "\n";

    // 随机访问tags（无需遍历整个结构）
    if (parsed->tags() && parsed->tags()->size() > 0) {
        std::cout << "First tag: " << parsed->tags()->Get(0)->c_str() << "\n";
    }
}

// 零拷贝网络传输示例
void flatbuffers_network_example() {
    // 发送端
    flatbuffers::FlatBufferBuilder builder(1024);
    auto name = builder.CreateString("Alice");
    auto email = builder.CreateString("alice@example.com");
    auto person = CreatePerson(builder, 1, name, email, 25);
    builder.Finish(person);

    // 直接发送buffer（零拷贝）
    uint8_t* data = builder.GetBufferPointer();
    size_t size = builder.GetSize();
    // send(socket, data, size, 0);

    // 接收端
    // recv(socket, buffer, size, 0);
    // 直接使用，无需解析！
    auto received = GetPerson(data);
    std::cout << "Received: " << received->name()->c_str() << "\n";
}
```

### 3.3 FlatBuffers vs Protobuf性能对比

```cpp
void benchmark_flatbuffers() {
    auto data = generate_test_data(10000);

    // 序列化
    auto start = std::chrono::high_resolution_clock::now();
    std::vector<std::vector<uint8_t>> serialized;

    for (const auto& p : data) {
        flatbuffers::FlatBufferBuilder builder(256);

        auto name = builder.CreateString(p.name);
        auto email = builder.CreateString(p.email);

        std::vector<flatbuffers::Offset<flatbuffers::String>> tags_vec;
        for (const auto& tag : p.tags) {
            tags_vec.push_back(builder.CreateString(tag));
        }
        auto tags = builder.CreateVector(tags_vec);

        auto person = CreatePerson(builder, p.id, name, email, p.age, tags);
        builder.Finish(person);

        // 复制buffer
        uint8_t* buf = builder.GetBufferPointer();
        size_t size = builder.GetSize();
        serialized.emplace_back(buf, buf + size);
    }

    auto serialize_time = std::chrono::high_resolution_clock::now() - start;

    // "反序列化"（实际上是零拷贝访问）
    start = std::chrono::high_resolution_clock::now();
    uint64_t sum = 0;
    for (const auto& buf : serialized) {
        auto person = GetPerson(buf.data());
        sum += person->id();  // 直接访问，无解析！
        // 注意：访问字符串会有轻微开销（验证偏移量）
        const char* name = person->name()->c_str();
        (void)name;
    }
    auto access_time = std::chrono::high_resolution_clock::now() - start;

    size_t total_size = 0;
    for (const auto& buf : serialized) {
        total_size += buf.size();
    }

    std::cout << "FlatBuffers:\n";
    std::cout << "  Serialize:   " << serialize_time.count() / 1e6 << " ms\n";
    std::cout << "  Access:      " << access_time.count() / 1e6 << " ms\n";
    std::cout << "  Total size:  " << total_size << " bytes\n";
    std::cout << "  Note: Access is NOT deserialization!\n";
}
```

### 4.2 Cap'n Proto特点

```cpp
#include <capnp/message.h>
#include <capnp/serialize.h>
#include "person.capnp.h"

// Cap'n Proto的核心优势：
// 1. 比FlatBuffers更快（编码即是数据）
// 2. 真正的零拷贝（无序列化步骤）
// 3. 强大的RPC支持

void capnproto_example() {
    // 创建消息
    ::capnp::MallocMessageBuilder message;
    Person::Builder person = message.initRoot<Person>();

    person.setId(1);
    person.setName("Alice");
    person.setEmail("alice@example.com");
    person.setAge(25);

    auto tags = person.initTags(3);
    tags.set(0, "tag1");
    tags.set(1, "tag2");
    tags.set(2, "tag3");

    // 序列化（实际上只是复制buffer）
    kj::Array<capnp::word> words = messageToFlatArray(message);
    kj::ArrayPtr<kj::byte> bytes = words.asBytes();

    // 反序列化（零拷贝！）
    kj::ArrayPtr<const capnp::word> word_array(
        reinterpret_cast<const capnp::word*>(bytes.begin()),
        bytes.size() / sizeof(capnp::word)
    );

    ::capnp::FlatArrayMessageReader reader(word_array);
    Person::Reader person_reader = reader.getRoot<Person>();

    std::cout << "ID: " << person_reader.getId() << "\n";
    std::cout << "Name: " << person_reader.getName().cStr() << "\n";
}

// Cap'n Proto的极致性能
void benchmark_capnproto() {
    const int N = 100000;

    auto start = std::chrono::high_resolution_clock::now();

    size_t total_size = 0;
    for (int i = 0; i < N; i++) {
        ::capnp::MallocMessageBuilder message;
        Person::Builder person = message.initRoot<Person>();

        person.setId(i);
        person.setName("Person");
        person.setEmail("person@example.com");
        person.setAge(25);

        // 获取大小（无实际序列化！）
        auto words = messageToFlatArray(message);
        total_size += words.asBytes().size();
    }

    auto time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "Cap'n Proto:\n";
    std::cout << "  Time:        " << time.count() / 1e6 << " ms\n";
    std::cout << "  Total size:  " << total_size << " bytes\n";
    std::cout << "  Throughput:  " << (double)N / (time.count() / 1e9) << " msg/s\n";
}
```

### 6.1 传统read/write的4次拷贝

```cpp
// 传统文件传输：4次拷贝，2次上下文切换
//
// 1. DMA: 磁盘 → 内核buffer (硬件拷贝)
// 2. CPU: 内核buffer → 用户buffer (read系统调用)
// 3. CPU: 用户buffer → socket buffer (write系统调用)
// 4. DMA: socket buffer → 网卡 (硬件拷贝)

void traditional_file_transfer(int socket_fd, const char* filename) {
    int fd = open(filename, O_RDONLY);
    char buffer[4096];

    ssize_t n;
    while ((n = read(fd, buffer, sizeof(buffer))) > 0) {
        // read: 内核 → 用户空间 (拷贝1)
        write(socket_fd, buffer, n);
        // write: 用户空间 → 内核 (拷贝2)
    }

    close(fd);
}
```

```cpp
// mmap: 将文件映射到内存地址空间
// 优势：
// 1. 减少拷贝（3次拷贝：磁盘→页缓存→用户空间合并为1次）
// 2. 延迟加载（按需分页）
// 3. 多进程共享内存

class MmapFile {
private:
    void* addr_;
    size_t size_;
    int fd_;

public:
    MmapFile(const char* filename, bool writable = false) {
        fd_ = open(filename, writable ? O_RDWR : O_RDONLY);
        if (fd_ < 0) {
            throw std::runtime_error("Cannot open file");
        }

        struct stat sb;
        if (fstat(fd_, &sb) < 0) {
            close(fd_);
            throw std::runtime_error("Cannot stat file");
        }

        size_ = sb.st_size;

        int prot = PROT_READ;
        if (writable) prot |= PROT_WRITE;

        addr_ = mmap(nullptr, size_, prot, MAP_SHARED, fd_, 0);
        if (addr_ == MAP_FAILED) {
            close(fd_);
            throw std::runtime_error("mmap failed");
        }
    }

    ~MmapFile() {
        if (addr_ != MAP_FAILED) {
            munmap(addr_, size_);
        }
        if (fd_ >= 0) {
            close(fd_);
        }
    }

    void* data() { return addr_; }
    size_t size() const { return size_; }

    // 建议内核预读
    void advise_sequential() {
        madvise(addr_, size_, MADV_SEQUENTIAL);
    }

    // 建议内核随机访问
    void advise_random() {
        madvise(addr_, size_, MADV_RANDOM);
    }

    // 预加载到内存
    void prefault() {
        madvise(addr_, size_, MADV_WILLNEED);
    }
};

void benchmark_mmap() {
```

```cpp
#include <fcntl.h>

// splice: 在两个文件描述符间移动数据（通过管道）
// 优势：完全零拷贝，可以用于任意fd

class PipePair {
private:
    int pipe_fds_[2];

public:
    PipePair() {
        if (pipe2(pipe_fds_, O_NONBLOCK) < 0) {
            throw std::runtime_error("pipe2 failed");
        }
    }

    ~PipePair() {
        close(pipe_fds_[0]);
        close(pipe_fds_[1]);
    }

    int read_end() const { return pipe_fds_[0]; }
    int write_end() const { return pipe_fds_[1]; }
};

// 使用splice实现零拷贝传输
ssize_t splice_transfer(int in_fd, int out_fd, size_t len) {
    PipePair pipe;

    // 数据流：in_fd → pipe → out_fd
    // 全程在内核空间，零拷贝！

    ssize_t total = 0;
    while (len > 0) {
        // in_fd → pipe
        ssize_t n = splice(in_fd, nullptr,
                          pipe.write_end(), nullptr,
                          len, SPLICE_F_MOVE | SPLICE_F_MORE);
        if (n <= 0) break;

        // pipe → out_fd
        ssize_t m = splice(pipe.read_end(), nullptr,
                          out_fd, nullptr,
                          n, SPLICE_F_MOVE | SPLICE_F_MORE);
        if (m <= 0) break;

        total += m;
        len -= m;
    }

    return total;
}
```

```cpp
// RDMA (Remote Direct Memory Access)
// 特点：
// 1. 绕过操作系统内核
// 2. 绕过CPU
// 3. 直接访问远程内存
// 4. 微秒级延迟（vs 毫秒级TCP）

#include <infiniband/verbs.h>

// RDMA连接示例（简化）
class RDMAConnection {
private:
    ibv_context* context_;
    ibv_pd* pd_;
    ibv_qp* qp_;
    ibv_cq* cq_;
    void* buffer_;
    ibv_mr* mr_;  // Memory Region

public:
    RDMAConnection(const char* device_name, size_t buffer_size) {
        // 1. 打开设备
        ibv_device** dev_list = ibv_get_device_list(nullptr);
        context_ = ibv_open_device(dev_list[0]);

        // 2. 分配Protection Domain
        pd_ = ibv_alloc_pd(context_);

        // 3. 分配buffer并注册
        buffer_ = malloc(buffer_size);
        mr_ = ibv_reg_mr(pd_, buffer_, buffer_size,
                        IBV_ACCESS_LOCAL_WRITE |
                        IBV_ACCESS_REMOTE_READ |
                        IBV_ACCESS_REMOTE_WRITE);

        // 4. 创建Completion Queue
        cq_ = ibv_create_cq(context_, 10, nullptr, nullptr, 0);

        // 5. 创建Queue Pair
        ibv_qp_init_attr qp_attr = {};
        qp_attr.send_cq = cq_;
        qp_attr.recv_cq = cq_;
        qp_attr.qp_type = IBV_QPT_RC;  // Reliable Connection
        qp_attr.cap.max_send_wr = 10;
        qp_attr.cap.max_recv_wr = 10;
        qp_attr.cap.max_send_sge = 1;
        qp_attr.cap.max_recv_sge = 1;

        qp_ = ibv_create_qp(pd_, &qp_attr);
    }

    // RDMA Write: 单边操作，无需远程CPU参与！
    void rdma_write(uint64_t remote_addr, uint32_t rkey, const void* data, size_t len) {
        memcpy(buffer_, data, len);

        ibv_sge sge = {};
        sge.addr = (uintptr_t)buffer_;
        sge.length = len;
        sge.lkey = mr_->lkey;

        ibv_send_wr wr = {};
        wr.wr_id = 0;
        wr.sg_list = &sge;
        wr.num_sge = 1;
        wr.opcode = IBV_WR_RDMA_WRITE;
        wr.send_flags = IBV_SEND_SIGNALED;
        wr.wr.rdma.remote_addr = remote_addr;
        wr.wr.rdma.rkey = rkey;

        ibv_send_wr* bad_wr;
        ibv_post_send(qp_, &wr, &bad_wr);

        // 等待完成
        ibv_wc wc;
        while (ibv_poll_cq(cq_, 1, &wc) == 0);
    }

    // RDMA Read: 单边读取
    void rdma_read(uint64_t remote_addr, uint32_t rkey, void* data, size_t len) {
        // 类似RDMA Write
    }

    ~RDMAConnection() {
        ibv_destroy_qp(qp_);
        ibv_destroy_cq(cq_);
        ibv_dereg_mr(mr_);
        free(buffer_);
        ibv_dealloc_pd(pd_);
        ibv_close_device(context_);
    }
};
```

## 15. 高性能RPC框架

### 15.1 完整RPC实现

```cpp
// 综合应用：FlatBuffers + 零拷贝 + 多路复用

class HighPerfRPCServer {
private:
    int epoll_fd_;
    std::unordered_map<int, ClientContext> clients_;

public:
    HighPerfRPCServer(int port) {
        // 创建监听socket
        int listen_fd = socket(AF_INET, SOCK_STREAM, 0);

        // 设置SO_REUSEADDR
        int opt = 1;
        setsockopt(listen_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

        // 绑定端口
        sockaddr_in addr = {};
        addr.sin_family = AF_INET;
        addr.sin_port = htons(port);
        addr.sin_addr.s_addr = INADDR_ANY;
        bind(listen_fd, (sockaddr*)&addr, sizeof(addr));

        listen(listen_fd, 128);

        // 创建epoll
        epoll_fd_ = epoll_create1(0);

        epoll_event ev = {};
        ev.events = EPOLLIN;
        ev.data.fd = listen_fd;
        epoll_ctl(epoll_fd_, EPOLL_CTL_ADD, listen_fd, &ev);
    }

    void run() {
        epoll_event events[128];

        while (true) {
            int n = epoll_wait(epoll_fd_, events, 128, -1);

            for (int i = 0; i < n; i++) {
                if (events[i].events & EPOLLIN) {
                    // 零拷贝接收
                    handle_request(events[i].data.fd);
                }
            }
        }
    }

private:
    void handle_request(int fd) {
        // 1. 接收FlatBuffers消息（零拷贝）
        char buffer[4096];
        ssize_t n = recv(fd, buffer, sizeof(buffer), 0);
        if (n <= 0) return;

        // 2. 直接解析（无反序列化！）
        auto request = GetRequest(buffer);

        // 3. 处理请求
        auto response = process(request);

        // 4. 零拷贝发送响应
        send(fd, response.data(), response.size(), 0);
    }

    std::vector<uint8_t> process(const Request* request) {
        // 业务逻辑
        flatbuffers::FlatBufferBuilder builder(256);

        // 构建响应
        auto response = CreateResponse(builder, 200, builder.CreateString("OK"));
        builder.Finish(response);

        // 返回buffer
        uint8_t* buf = builder.GetBufferPointer();
        size_t size = builder.GetSize();
        return std::vector<uint8_t>(buf, buf + size);
    }
};
```
