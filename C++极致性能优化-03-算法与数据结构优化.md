## 1. Cache-friendly算法设计

### 1.1 顺序访问 vs 随机访问

```cpp
#include <iostream>
#include <chrono>
#include <vector>
#include <random>

// 不好：随机访问（Cache miss多）
long long sum_random_access(const std::vector<int>& data,
                            const std::vector<int>& indices) {
    long long sum = 0;
    for (int idx : indices) {
        sum += data[idx];  // 随机访问，每次可能Cache miss
    }
    return sum;
}

// 好：排序后顺序访问
long long sum_sorted_access(const std::vector<int>& data,
                            std::vector<int> indices) {
    // 1. 排序索引
    std::sort(indices.begin(), indices.end());

    // 2. 顺序访问（Cache friendly）
    long long sum = 0;
    for (int idx : indices) {
        sum += data[idx];
    }
    return sum;
}

// 性能测试
void benchmark_cache_friendly() {
    constexpr int N = 10000000;
    std::vector<int> data(N);
    std::vector<int> indices(1000000);

    // 初始化
    for (int i = 0; i < N; i++) {
        data[i] = i;
    }

    std::mt19937 gen(42);
    std::uniform_int_distribution<> dis(0, N - 1);
    for (int& idx : indices) {
        idx = dis(gen);
    }

    // 测试随机访问
    auto start = std::chrono::high_resolution_clock::now();
    long long sum1 = sum_random_access(data, indices);
    auto end = std::chrono::high_resolution_clock::now();
    auto random_time = std::chrono::duration_cast<std::chrono::milliseconds>(
        end - start).count();

    // 测试排序访问
    start = std::chrono::high_resolution_clock::now();
    long long sum2 = sum_sorted_access(data, indices);
    end = std::chrono::high_resolution_clock::now();
    auto sorted_time = std::chrono::duration_cast<std::chrono::milliseconds>(
        end - start).count();

    std::cout << "Random access: " << random_time << " ms\n";
    std::cout << "Sorted access: " << sorted_time << " ms\n";
    std::cout << "Speedup:       " << (double)random_time / sorted_time << "x\n";
}

// 输出示例：
// Random access: 850 ms
// Sorted access: 120 ms（包含排序时间）
// Speedup:       7.1x
```

```cpp
struct Node {
    int data;
    Node* next;  // 指针跳跃，Cache miss多

class LinkedList {
    Node* head;

public:
    long long sum() {

        }
        return total;
    }
};
```

### 1.3 二叉树 vs B树

```cpp
// 二叉搜索树（Cache不友好）
struct BSTNode {
    int key;
    BSTNode *left, *right;  // 两个指针跳跃
};

// B树（Cache友好）
template<int B = 64>  // 块大小 = Cache line大小
struct BTreeNode {
    int num_keys;
    int keys[B];          // 连续存储
    BTreeNode* children[B + 1];
};

// 搜索性能对比
bool bst_search(BSTNode* root, int key) {
    if (!root) return false;
    if (root->key == key) return true;

    // 每次比较可能Cache miss（指针跳跃）
    return key < root->key ?
        bst_search(root->left, key) :
        bst_search(root->right, key);
}

template<int B>
bool btree_search(BTreeNode<B>* node, int key) {
    if (!node) return false;

    // 在节点内二分查找（连续内存，Cache友好）
    int pos = std::lower_bound(node->keys,
                               node->keys + node->num_keys,
                               key) - node->keys;

    if (pos < node->num_keys && node->keys[pos] == key) {
        return true;
    }

    return btree_search(node->children[pos], key);
}

// B树优势：
// - 更少的指针跳跃（高扇出）
// - 节点内顺序访问（Cache友好）
// - 更少的Cache miss
// 性能提升：5-10倍
```

## 2. 位运算优化技巧

### 2.1 基础位运算

```cpp
// 1. 检查奇偶
bool is_even_slow(int x) {
    return x % 2 == 0;  // 除法指令，~20周期
}

bool is_even_fast(int x) {
    return (x & 1) == 0;  // 位运算，1周期
}

// 2. 2的幂次判断
bool is_power_of_2_slow(int x) {
    if (x <= 0) return false;
    while (x > 1) {
        if (x % 2 != 0) return false;
        x /= 2;
    }
    return true;
}

bool is_power_of_2_fast(int x) {
    return x > 0 && (x & (x - 1)) == 0;
}

// 3. 向上取整到2的幂
int round_up_power_of_2_slow(int x) {
    int result = 1;
    while (result < x) {
        result *= 2;
    }
    return result;
}

int round_up_power_of_2_fast(int x) {
    x--;
    x |= x >> 1;
    x |= x >> 2;
    x |= x >> 4;
    x |= x >> 8;
    x |= x >> 16;
    return x + 1;
}

// 4. 计数置位（Population Count）
int popcount_slow(uint32_t x) {
    int count = 0;
    while (x) {
        if (x & 1) count++;
        x >>= 1;
    }
    return count;
}

int popcount_fast(uint32_t x) {
    return __builtin_popcount(x);  // POPCNT指令
}

// 5. 前导零（Count Leading Zeros）
int clz_slow(uint32_t x) {
    if (x == 0) return 32;
    int count = 0;
    while ((x & 0x80000000) == 0) {
        count++;
        x <<= 1;
    }
    return count;
}

int clz_fast(uint32_t x) {
    return __builtin_clz(x);  // LZCNT指令
}

// 6. 后尾零（Count Trailing Zeros）
int ctz_slow(uint32_t x) {
    if (x == 0) return 32;
    int count = 0;
    while ((x & 1) == 0) {
        count++;
        x >>= 1;
    }
    return count;
}

int ctz_fast(uint32_t x) {
    return __builtin_ctz(x);  // TZCNT指令
}
```

```cpp
void swap_xor(int& a, int& b) {

        a ^= b;
        b ^= a;
        a ^= b;

bool same_sign(int a, int b) {

// 应用4：绝对值（无分支）
int abs_branchless(int x) {
    int mask = x >> 31;  // 全0（正数）或全1（负数）
    return (x ^ mask) - mask;
}

// 应用5：最小值/最大值（无分支）
int min_branchless(int a, int b) {
    return b ^ ((a ^ b) & -(a < b));
}

int max_branchless(int a, int b) {
    return a ^ ((a ^ b) & -(a < b));
}

// 应用6：条件设置（无分支）
int conditional_set(bool condition, int true_val, int false_val) {
    return false_val ^ (-condition & (true_val ^ false_val));
}

uint32_t reverse_bits(uint32_t x) {
    x = ((x & 0xAAAAAAAA) >> 1) | ((x & 0x55555555) << 1);
    x = ((x & 0xCCCCCCCC) >> 2) | ((x & 0x33333333) << 2);
    x = ((x & 0xF0F0F0F0) >> 4) | ((x & 0x0F0F0F0F) << 4);
    x = ((x & 0xFF00FF00) >> 8) | ((x & 0x00FF00FF) << 8);
    x = (x >> 16) | (x << 16);
    return x;
}

    uint32_t mask = ((1U << length) - 1) << start;
```

## 3. SWAR技术

### 3.1 SWAR基础

**SWAR (SIMD Within A Register)**: 在一个寄存器内并行处理多个数据

```cpp
// 示例：并行计数4个字节中的置位
uint32_t popcount_4bytes_scalar(uint32_t x) {
    int count = 0;
    for (int i = 0; i < 4; i++) {
        count += __builtin_popcount((x >> (i * 8)) & 0xFF);
    }
    return count;
}

uint32_t popcount_4bytes_swar(uint32_t x) {
    // 并行计算4个字节的popcount
    x = x - ((x >> 1) & 0x55555555);
    x = (x & 0x33333333) + ((x >> 2) & 0x33333333);
    x = (x + (x >> 4)) & 0x0F0F0F0F;
    x = x + (x >> 8);
    x = x + (x >> 16);
    return x & 0x3F;
}

// 性能提升：3-4倍
```

### 3.2 SWAR应用：字符串操作

```cpp
// 查找字符串中的空字节（C字符串结束符）
bool has_zero_byte_scalar(uint64_t x) {
    for (int i = 0; i < 8; i++) {
        if (((x >> (i * 8)) & 0xFF) == 0) {
            return true;
        }
    }
    return false;
}

bool has_zero_byte_swar(uint64_t x) {
    // SWAR魔法：并行检测8个字节
    return ((x - 0x0101010101010101ULL) &
            ~x &
            0x8080808080808080ULL) != 0;
}

// strlen实现（SWAR加速）
size_t strlen_swar(const char* str) {
    const char* s = str;

    // 对齐到8字节边界
    while ((uintptr_t)s & 7) {
        if (*s == '\0') return s - str;
        s++;
    }

    // 每次处理8字节
    const uint64_t* p = (const uint64_t*)s;
    while (!has_zero_byte_swar(*p)) {
        p++;
    }

    // 精确定位
    s = (const char*)p;
    while (*s != '\0') s++;

    return s - str;
}

// 性能提升：2-3倍（相比标准strlen）
```

### 3.3 SWAR应用：并行比较

```cpp
// 并行比较4个整数
bool all_equal_scalar(uint32_t a, uint32_t b, uint32_t c, uint32_t d) {
    return a == b && b == c && c == d;
}

bool all_equal_swar(uint32_t a, uint32_t b, uint32_t c, uint32_t d) {
    // 将4个32位整数视为16个字节
    uint64_t ab = ((uint64_t)a << 32) | b;
    uint64_t cd = ((uint64_t)c << 32) | d;

    // XOR检测差异
    uint64_t diff = ab ^ cd;

    // 如果全相等，diff应该为0
    return diff == 0 && ((a ^ b) == 0);
}

// 查找字符
const char* strchr_swar(const char* str, char c) {
    const char* s = str;
    uint64_t pattern = 0x0101010101010101ULL * (unsigned char)c;

    // 对齐到8字节
    while ((uintptr_t)s & 7) {
        if (*s == c) return s;
        if (*s == '\0') return nullptr;
        s++;
    }

    const uint64_t* p = (const uint64_t*)s;
    while (true) {
        uint64_t word = *p;

        // 检查是否有匹配的字符
        uint64_t match = word ^ pattern;
        if (has_zero_byte_swar(match)) {
            // 找到匹配
            s = (const char*)p;
            while (*s != c) s++;
            return s;
        }

        // 检查是否到达字符串末尾
        if (has_zero_byte_swar(word)) {
            return nullptr;
        }

        p++;
    }
}
```

## 4. 哈希表优化

### 4.1 开放寻址 vs 链式

```cpp
// 链式哈希表（标准unordered_map）
// 问题：
// - 指针跳跃，Cache不友好
// - 内存碎片
// - 动态分配开销

// 开放寻址哈希表（Robin Hood Hashing）
template<typename K, typename V>
class RobinHoodHashMap {
    struct Entry {
        K key;
        V value;
        int distance;  // 距离理想位置的距离
        bool occupied;
    };

    std::vector<Entry> table;
    size_t size_;

    size_t hash(const K& key) const {
        return std::hash<K>{}(key) % table.size();
    }

public:
    RobinHoodHashMap(size_t capacity) : size_(0) {
        table.resize(capacity);
        for (auto& e : table) {
            e.occupied = false;
        }
    }

    void insert(const K& key, const V& value) {
        size_t idx = hash(key);
        Entry entry{key, value, 0, true};

        while (true) {
            if (!table[idx].occupied) {
                table[idx] = entry;
                size_++;
                return;
            }

            // Robin Hood规则：如果当前entry比位置上的entry更"穷"
            // （距离更远），则交换
            if (entry.distance > table[idx].distance) {
                std::swap(entry, table[idx]);
            }

            entry.distance++;
            idx = (idx + 1) % table.size();
        }
    }

    V* find(const K& key) {
        size_t idx = hash(key);
        int distance = 0;

        while (table[idx].occupied) {
            if (table[idx].key == key) {
                return &table[idx].value;
            }

            // 如果距离超过了理论最大值，说明不存在
            if (distance > table[idx].distance) {
                return nullptr;
            }

            distance++;
            idx = (idx + 1) % table.size();
        }

        return nullptr;
    }
};

// 优势：
// - 连续内存，Cache友好
// - 无指针跳跃
// - 更均匀的探测距离
// - 性能提升：2-3倍
```

### 4.2 Swiss Table（Google's absl::flat_hash_map）

```cpp
// Swiss Table核心思想：
// - 使用SIMD并行查找
// - 元数据与数据分离
// - 每个slot有一个control byte（元数据）

template<typename K, typename V>
class SwissTable {
    static constexpr size_t GROUP_SIZE = 16;  // SIMD宽度

    struct alignas(16) ControlGroup {
        int8_t ctrl[GROUP_SIZE];
        // ctrl[i] = -1: empty
        // ctrl[i] = -2: deleted
        // ctrl[i] >= 0: hash的高7位
    };

    std::vector<ControlGroup> control;
    std::vector<std::pair<K, V>> slots;

    // H2哈希：取高7位作为fingerprint
    int8_t h2_hash(size_t hash) const {
        return (hash >> 57) & 0x7F;
    }

public:
    V* find(const K& key) {
        size_t hash = std::hash<K>{}(key);
        int8_t h2 = h2_hash(hash);
        size_t group_idx = hash % control.size();

        // SIMD并行匹配
        #ifdef __SSE2__
        __m128i target = _mm_set1_epi8(h2);

        const ControlGroup& group = control[group_idx];
        __m128i ctrl = _mm_load_si128((__m128i*)group.ctrl);

        // 并行比较16个control byte
        __m128i cmp = _mm_cmpeq_epi8(ctrl, target);
        int mask = _mm_movemask_epi8(cmp);

        // 遍历匹配的位置
        while (mask != 0) {
            int bit = __builtin_ctz(mask);
            size_t slot_idx = group_idx * GROUP_SIZE + bit;

            if (slots[slot_idx].first == key) {
                return &slots[slot_idx].second;
            }

            mask &= mask - 1;  // 清除最低位
        }
        #endif

        return nullptr;
    }
};

// 优势：
// - SIMD并行查找（16路）
// - 元数据紧凑，Cache友好
// - 查找性能：比std::unordered_map快2-4倍
```

### 4.3 Perfect Hashing

```cpp
// 场景：键集合已知且固定
// 使用完美哈希：无碰撞，O(1)查找

class PerfectHashMap {
    // 两级哈希
    struct SecondLevel {
        int a, b, size;
        std::vector<int> values;
    };

    std::vector<SecondLevel> table;
    int n;  // 键的数量

    int hash1(int key) const {
        return key % n;
    }

    int hash2(int key, int a, int b, int size) const {
        return ((a * key + b) % n) % size;
    }

public:
    // 构造：需要所有键
    void build(const std::vector<int>& keys) {
        n = keys.size();
        table.resize(n);

        // 第一级：分组
        std::vector<std::vector<int>> buckets(n);
        for (int key : keys) {
            buckets[hash1(key)].push_back(key);
        }

        // 第二级：为每个桶构建无碰撞哈希
        for (int i = 0; i < n; i++) {
            if (buckets[i].empty()) continue;

            int bucket_size = buckets[i].size();
            int table_size = bucket_size * bucket_size;

            table[i].size = table_size;
            table[i].values.resize(table_size, -1);

            // 尝试找到无碰撞的参数
            bool found = false;
            for (int a = 1; a < n && !found; a++) {
                for (int b = 0; b < n && !found; b++) {
                    std::vector<bool> used(table_size, false);
                    bool collision = false;

                    for (int key : buckets[i]) {
                        int idx = hash2(key, a, b, table_size);
                        if (used[idx]) {
                            collision = true;
                            break;
                        }
                        used[idx] = true;
                    }

                    if (!collision) {
                        table[i].a = a;
                        table[i].b = b;
                        found = true;

                        // 填充值
                        for (int key : buckets[i]) {
                            int idx = hash2(key, a, b, table_size);
                            table[i].values[idx] = key;
                        }
                    }
                }
            }
        }
    }

    bool find(int key) const {
        int bucket = hash1(key);
        if (table[bucket].values.empty()) return false;

        int idx = hash2(key, table[bucket].a, table[bucket].b, table[bucket].size);
        return table[bucket].values[idx] == key;
    }
};

// 优势：
// - 真正的O(1)查找，无碰撞
// - 预测性能稳定
// - 适用于静态键集合
```

## 5. 排序算法优化

### 5.1 小数组优化

```cpp
// 插入排序：小数组（<20）最快
template<typename T>
void insertion_sort(T* arr, int n) {
    for (int i = 1; i < n; i++) {
        T key = arr[i];
        int j = i - 1;

        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;
    }
}

// 优化的快速排序：混合算法
template<typename T>
void quick_sort_optimized(T* arr, int left, int right) {
    while (right - left > 20) {  // 大数组用快排
        // 三数取中选择pivot
        int mid = left + (right - left) / 2;
        if (arr[mid] < arr[left]) std::swap(arr[left], arr[mid]);
        if (arr[right] < arr[left]) std::swap(arr[left], arr[right]);
        if (arr[right] < arr[mid]) std::swap(arr[mid], arr[right]);

        T pivot = arr[mid];
        std::swap(arr[mid], arr[right - 1]);

        int i = left, j = right - 1;
        while (true) {
            while (arr[++i] < pivot);
            while (arr[--j] > pivot);
            if (i >= j) break;
            std::swap(arr[i], arr[j]);
        }
        std::swap(arr[i], arr[right - 1]);

        // 尾递归优化：只递归较小的一半
        if (i - left < right - i) {
            quick_sort_optimized(arr, left, i - 1);
            left = i + 1;
        } else {
            quick_sort_optimized(arr, i + 1, right);
            right = i - 1;
        }
    }

    // 小数组用插入排序
    insertion_sort(arr + left, right - left + 1);
}

// 性能提升：15-25%（相比标准快排）
```

### 5.2 基数排序（整数专用）

```cpp
// LSD基数排序（整数）
void radix_sort_lsd(std::vector<uint32_t>& arr) {
    constexpr int BITS = 8;  // 每次处理8位
    constexpr int BUCKETS = 1 << BITS;  // 256个桶

    std::vector<uint32_t> temp(arr.size());
    int count[BUCKETS];

    // 4轮，每轮8位
    for (int shift = 0; shift < 32; shift += BITS) {
        // 计数
        std::fill(count, count + BUCKETS, 0);
        for (uint32_t val : arr) {
            int bucket = (val >> shift) & (BUCKETS - 1);
            count[bucket]++;
        }

        // 前缀和
        for (int i = 1; i < BUCKETS; i++) {
            count[i] += count[i - 1];
        }

        // 分配
        for (int i = arr.size() - 1; i >= 0; i--) {
            int bucket = (arr[i] >> shift) & (BUCKETS - 1);
            temp[--count[bucket]] = arr[i];
        }

        arr = temp;
    }
}

// 性能：O(n)，对于32位整数，性能比快排快2-3倍
```

```cpp
// 双调排序（Bitonic Sort）- SIMD友好
void bitonic_sort_simd(float* arr, int n) {
    // 要求n是2的幂
    for (int k = 2; k <= n; k *= 2) {
        for (int j = k / 2; j > 0; j /= 2) {
            for (int i = 0; i < n; i++) {
                int ixj = i ^ j;
                if (ixj > i) {
                    if ((i & k) == 0) {
                        if (arr[i] > arr[ixj]) {
                            std::swap(arr[i], arr[ixj]);
                        }
                    } else {
                        if (arr[i] < arr[ixj]) {
                            std::swap(arr[i], arr[ixj]);
                        }
                    }
                }
            }
        }
    }
}

// SIMD排序优势：
// - 并行比较交换
// - 无分支
// - 性能提升：2-4倍（小数组）
```

```cpp
bool strcmp_simd(const char* s1, const char* s2, size_t n) {
    size_t i = 0;

    // 每次比较32字节
    for (; i + 32 <= n; i += 32) {
        __m256i v1 = _mm256_loadu_si256((__m256i*)(s1 + i));
        __m256i v2 = _mm256_loadu_si256((__m256i*)(s2 + i));

        __m256i cmp = _mm256_cmpeq_epi8(v1, v2);
        int mask = _mm256_movemask_epi8(cmp);

        if (mask != -1) {  // 有不相等的字节
            return false;
        }
    }

    // 处理剩余
    for (; i < n; i++) {
        if (s1[i] != s2[i]) return false;
    }

    return true;
}
```

### 6.2 Boyer-Moore字符串搜索

```cpp
// 优化的Boyer-Moore算法
class BoyerMoore {
    std::string pattern;
    int bad_char[256];
    std::vector<int> good_suffix;

    void preprocess_bad_char() {
        std::fill(bad_char, bad_char + 256, -1);
        for (int i = 0; i < pattern.size(); i++) {
            bad_char[(unsigned char)pattern[i]] = i;
        }
    }

    void preprocess_good_suffix() {
        int m = pattern.size();
        good_suffix.resize(m + 1);
        std::vector<int> suffix(m + 1);

        // 计算suffix数组
        suffix[m] = m + 1;
        int j = m + 1;
        for (int i = m; i > 0; i--) {
            if (i > j && suffix[i + m - 1] < i - j) {
                suffix[i] = suffix[i + m - 1];
            } else {
                if (i < j) j = i;
                while (j > 0 && pattern[j - 1] == pattern[j + m - 1 - i]) {
                    j--;
                }
                suffix[i] = i - j;
            }
        }

        // 计算good_suffix数组
        std::fill(good_suffix.begin(), good_suffix.end(), m);
        for (int i = 0; i < m; i++) {
            good_suffix[m - suffix[i]] = m - i;
        }
    }

public:
    BoyerMoore(const std::string& pat) : pattern(pat) {
        preprocess_bad_char();
        preprocess_good_suffix();
    }

    int search(const std::string& text) {
        int n = text.size();
        int m = pattern.size();

        int i = 0;
        while (i <= n - m) {
            int j = m - 1;

            // 从右向左匹配
            while (j >= 0 && pattern[j] == text[i + j]) {
                j--;
            }

            if (j < 0) {
                return i;  // 找到匹配
            }

            // 计算跳跃距离
            int bad_char_shift = j - bad_char[(unsigned char)text[i + j]];
            int good_suffix_shift = good_suffix[j + 1];

            i += std::max(bad_char_shift, good_suffix_shift);
        }

        return -1;  // 未找到
    }
};

// 性能：平均O(n/m)，最坏O(nm)
// 在长模式串时比KMP快3-5倍
```

### 7.1 二分查找优化

```cpp
// 标准二分查找
int binary_search_standard(const int* arr, int n, int target) {
    int left = 0, right = n - 1;

    while (left <= right) {
        int mid = left + (right - left) / 2;

        if (arr[mid] == target) {
            return mid;
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }

    return -1;
}

// 分支消除版本
int binary_search_branchless(const int* arr, int n, int target) {
    int left = 0;
    int right = n;

    while (left < right) {
        int mid = left + (right - left) / 2;

        // 无分支：使用条件移动
        left = (arr[mid] < target) ? mid + 1 : left;
        right = (arr[mid] >= target) ? mid : right;
    }

    return (left < n && arr[left] == target) ? left : -1;
}

// Eytzinger布局（Cache友好）
class EytzingerArray {
    std::vector<int> data;
    int size_;

    // 将排序数组转换为Eytzinger布局
    void build(const int* arr, int n, int i = 0, int k = 1) {
        if (k <= n) {
            build(arr, n, i, 2 * k);
            data[k] = arr[i];
            build(arr, n, i + 1, 2 * k + 1);
        }
    }

public:
    EytzingerArray(const int* arr, int n) : size_(n) {
        data.resize(n + 1);
        build(arr, n);
    }

    int search(int target) const {
        int k = 1;

        while (k <= size_) {
            // 预取下一层
            __builtin_prefetch(&data[2 * k], 0, 3);
            __builtin_prefetch(&data[2 * k + 1], 0, 3);

            k = 2 * k + (data[k] < target);
        }

        k >>= __builtin_ffs(~k);
        return k == 0 ? -1 : k - 1;
    }
};

// Eytzinger优势：
// - 更好的Cache locality
// - 预取友好
// - 性能提升：30-50%（大数组）
```

```cpp
// 对于均匀分布的数据，插值搜索更快
int interpolation_search(const int* arr, int n, int target) {
    int left = 0, right = n - 1;

    while (left <= right &&
           target >= arr[left] &&
           target <= arr[right]) {

        if (left == right) {
            return arr[left] == target ? left : -1;
        }

        // 插值估计位置
        int pos = left + ((double)(right - left) /
                         (arr[right] - arr[left])) *
                         (target - arr[left]);

        if (arr[pos] == target) {
            return pos;
        } else if (arr[pos] < target) {
            left = pos + 1;
        } else {
            right = pos - 1;
        }
    }

    return -1;
}

// 性能：O(log log n)（均匀分布）
// 比二分快2-3倍（均匀数据）
```

```cpp
// 标准转置（Cache不友好）
void transpose_standard(int** src, int** dst, int n) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            dst[j][i] = src[i][j];  // 列写入，Cache miss多
        }
    }
}

// Cache-oblivious转置（分治）
void transpose_recursive(int** src, int** dst,
                        int row, int col,
                        int n_rows, int n_cols) {
    if (n_rows <= 32 && n_cols <= 32) {
        // 基础情况：小块直接转置
        for (int i = 0; i < n_rows; i++) {
            for (int j = 0; j < n_cols; j++) {
                dst[col + j][row + i] = src[row + i][col + j];
            }
        }
    } else if (n_rows >= n_cols) {
        // 按行分割
        transpose_recursive(src, dst, row, col, n_rows / 2, n_cols);
        transpose_recursive(src, dst, row + n_rows / 2, col,
                          n_rows - n_rows / 2, n_cols);
    } else {
        // 按列分割
        transpose_recursive(src, dst, row, col, n_rows, n_cols / 2);
        transpose_recursive(src, dst, row, col + n_cols / 2,
                          n_rows, n_cols - n_cols / 2);
    }
}

// 性能提升：3-5倍（大矩阵）
```

```cpp
// Strassen算法（Cache-oblivious）
void matmul_strassen(double** A, double** B, double** C,
                    int n, int row_a, int col_a,
                    int row_b, int col_b,
                    int row_c, int col_c) {
    if (n <= 64) {
        // 基础情况：直接乘法
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    C[row_c + i][col_c + j] +=
                        A[row_a + i][col_a + k] * B[row_b + k][col_b + j];
                }
            }
        }
    } else {
        // Strassen分治
        int m = n / 2;

        // 7次递归调用
        // M1 = (A11 + A22) * (B11 + B22)
        // M2 = (A21 + A22) * B11
        // ... 等等

        // 合并结果
        // C11 = M1 + M4 - M5 + M7
        // C12 = M3 + M5
        // C21 = M2 + M4
        // C22 = M1 - M2 + M3 + M6
    }
}
```
