### 1.1 Bloom Filter（布隆过滤器）

#### 1.1.1 基本原理

Bloom Filter是一个节省空间的概率型数据结构，用于测试元素是否在集合中。

**特点：**
- 可能误报（false positive），但不会漏报（no false negative）
- 空间效率极高
- 查询速度O(k)，k为哈希函数数量

```cpp
#include <vector>
#include <functional>
#include <cmath>

class BloomFilter {
private:
    std::vector<bool> bits_;
    size_t num_bits_;
    size_t num_hashes_;

    // 多个哈希函数（使用双哈希技巧）
    size_t hash(const std::string& item, size_t seed) const {
        size_t h1 = std::hash<std::string>{}(item);
        size_t h2 = std::hash<std::string>{}(item + std::to_string(seed));
        return (h1 + seed * h2) % num_bits_;
    }

public:
    // n: 预期元素数量, fpr: 期望误报率
    BloomFilter(size_t n, double fpr = 0.01) {
        // 最优位数：m = -(n * ln(fpr)) / (ln(2)^2)
        num_bits_ = static_cast<size_t>(
            -n * std::log(fpr) / (std::log(2) * std::log(2))
        );

        // 最优哈希数：k = (m / n) * ln(2)
        num_hashes_ = static_cast<size_t>(
            (static_cast<double>(num_bits_) / n) * std::log(2)
        );

        bits_.resize(num_bits_, false);
    }

    void insert(const std::string& item) {
        for (size_t i = 0; i < num_hashes_; i++) {
            size_t pos = hash(item, i);
            bits_[pos] = true;
        }
    }

    bool contains(const std::string& item) const {
        for (size_t i = 0; i < num_hashes_; i++) {
            size_t pos = hash(item, i);
            if (!bits_[pos]) {
                return false;  // 确定不存在
            }
        }
        return true;  // 可能存在（可能误报）
    }

    size_t memory_usage() const {
        return num_bits_ / 8;  // 字节数
    }
};

// 使用示例
void bloom_filter_example() {
    // 100万元素，1%误报率
    BloomFilter filter(1000000, 0.01);

    // 插入元素
    for (int i = 0; i < 1000000; i++) {
        filter.insert("user_" + std::to_string(i));
    }

    // 查询
    bool exists = filter.contains("user_12345");  // 可能存在
    bool not_exists = filter.contains("user_999999999");  // 确定不存在

    std::cout << "Memory usage: " << filter.memory_usage() << " bytes\n";
    // vs std::unordered_set: 需要~40MB
    // Bloom Filter: 只需~1.2MB (节省97%)
}
```

#### 1.1.2 SIMD优化的Bloom Filter

```cpp
#include <immintrin.h>

class SIMDBloomFilter {
private:
    alignas(32) std::vector<uint32_t> bits_;
    size_t num_bits_;
    static constexpr size_t K = 8;  // 8个哈希函数

    // 使用AVX2并行计算8个哈希
    void compute_hashes_avx2(const std::string& item, uint32_t* hashes) const {
        uint64_t h1 = std::hash<std::string>{}(item);
        uint64_t h2 = h1 >> 32;

        // 向量化计算：h = h1 + i * h2
        __m256i h1_vec = _mm256_set1_epi32(h1 & 0xFFFFFFFF);
        __m256i h2_vec = _mm256_set1_epi32(h2 & 0xFFFFFFFF);
        __m256i i_vec = _mm256_setr_epi32(0, 1, 2, 3, 4, 5, 6, 7);

        __m256i hash_vec = _mm256_add_epi32(h1_vec, _mm256_mullo_epi32(i_vec, h2_vec));

        _mm256_storeu_si256((__m256i*)hashes, hash_vec);
    }

public:
    SIMDBloomFilter(size_t n, double fpr = 0.01) {
        num_bits_ = static_cast<size_t>(
            -n * std::log(fpr) / (std::log(2) * std::log(2))
        );

        // 向上取整到32的倍数（AVX2）
        num_bits_ = (num_bits_ + 31) & ~31;

        bits_.resize(num_bits_ / 32, 0);
    }

    void insert(const std::string& item) {
        uint32_t hashes[K];
        compute_hashes_avx2(item, hashes);

        for (size_t i = 0; i < K; i++) {
            uint32_t h = hashes[i] % num_bits_;
            uint32_t word_idx = h / 32;
            uint32_t bit_idx = h % 32;
            bits_[word_idx] |= (1u << bit_idx);
        }
    }

    bool contains(const std::string& item) const {
        uint32_t hashes[K];
        compute_hashes_avx2(item, hashes);

        for (size_t i = 0; i < K; i++) {
            uint32_t h = hashes[i] % num_bits_;
            uint32_t word_idx = h / 32;
            uint32_t bit_idx = h % 32;

            if (!(bits_[word_idx] & (1u << bit_idx))) {
                return false;
            }
        }

        return true;
    }
};

// 性能提升：2-3倍（相比标量版本）
```

### 1.2 Count-Min Sketch

Count-Min Sketch用于估计流数据中元素的频率，空间复杂度极低。

```cpp
class CountMinSketch {
private:
    std::vector<std::vector<int>> table_;
    size_t width_;   // 列数
    size_t depth_;   // 行数（哈希函数数量）

    size_t hash(const std::string& item, size_t seed) const {
        size_t h = std::hash<std::string>{}(item + std::to_string(seed));
        return h % width_;
    }

public:
    // width: 列数, depth: 哈希函数数量
    // 误差界限: ε = e / width
    // 置信度: δ = 1 - (1/e)^depth
    CountMinSketch(size_t width, size_t depth)
        : width_(width), depth_(depth) {
        table_.resize(depth_, std::vector<int>(width_, 0));
    }

    void add(const std::string& item, int count = 1) {
        for (size_t i = 0; i < depth_; i++) {
            size_t pos = hash(item, i);
            table_[i][pos] += count;
        }
    }

    int estimate(const std::string& item) const {
        int min_count = std::numeric_limits<int>::max();

        for (size_t i = 0; i < depth_; i++) {
            size_t pos = hash(item, i);
            min_count = std::min(min_count, table_[i][pos]);
        }

        return min_count;  // 返回最小值（保守估计）
    }

    size_t memory_usage() const {
        return width_ * depth_ * sizeof(int);
    }
};

// 使用示例：网站访问统计
void count_min_example() {
    // 1000列，5行，误差约0.27%，置信度99.3%
    CountMinSketch cms(1000, 5);

    // 模拟流数据
    std::vector<std::string> urls = {
        "/home", "/about", "/products", "/home", "/home"
    };

    for (const auto& url : urls) {
        cms.add(url);
    }

    // 查询频率
    int home_count = cms.estimate("/home");  // ~3
    int about_count = cms.estimate("/about");  // ~1

    std::cout << "Memory usage: " << cms.memory_usage() << " bytes\n";
    // vs std::unordered_map<string, int>: 节省90-99%内存
}
```

### 1.3 HyperLogLog（基数统计）

HyperLogLog用于估计大数据集的基数（distinct count），误差<2%，内存使用极少。

```cpp
#include <cmath>
#include <vector>
#include <algorithm>

class HyperLogLog {
private:
    std::vector<uint8_t> registers_;
    size_t m_;  // 寄存器数量 (2^p)
    size_t p_;  // 精度参数

    // 计算前导零个数
    uint8_t leading_zeros(uint64_t hash, uint8_t max_bits) const {
        if (hash == 0) return max_bits;

        uint8_t count = 0;
        while ((hash & (1ULL << (63 - count))) == 0 && count < max_bits) {
            count++;
        }
        return count + 1;
    }

public:
    explicit HyperLogLog(size_t precision = 14) : p_(precision) {
        m_ = 1 << p_;  // 2^p个寄存器
        registers_.resize(m_, 0);
    }

    void add(const std::string& item) {
        uint64_t hash = std::hash<std::string>{}(item);

        // 前p位决定寄存器索引
        size_t j = hash & ((1 << p_) - 1);

        // 剩余位计算前导零
        uint64_t w = hash >> p_;
        uint8_t lz = leading_zeros(w, 64 - p_);

        // 更新寄存器（取最大值）
        registers_[j] = std::max(registers_[j], lz);
    }

    size_t estimate() const {
        double raw_estimate = alpha_m() * m_ * m_ / sum_of_powers();

        // 小范围修正
        if (raw_estimate <= 2.5 * m_) {
            size_t zeros = std::count(registers_.begin(), registers_.end(), 0);
            if (zeros != 0) {
                return m_ * std::log(static_cast<double>(m_) / zeros);
            }
        }

        // 大范围修正
        if (raw_estimate <= (1.0/30.0) * (1ULL << 32)) {
            return static_cast<size_t>(raw_estimate);
        } else {
            return static_cast<size_t>(
                -(1ULL << 32) * std::log(1 - raw_estimate / (1ULL << 32))
            );
        }
    }

    size_t memory_usage() const {
        return m_;  // 每个寄存器1字节
    }

private:
    double alpha_m() const {
        switch (m_) {
            case 16: return 0.673;
            case 32: return 0.697;
            case 64: return 0.709;
            default: return 0.7213 / (1 + 1.079 / m_);
        }
    }

    double sum_of_powers() const {
        double sum = 0;
        for (uint8_t reg : registers_) {
            sum += std::pow(2, -reg);
        }
        return sum;
    }
};

// 使用示例：统计独立访客
void hyperloglog_example() {
    HyperLogLog hll(14);  // 精度14，内存16KB，误差~0.8%

    // 插入100万个唯一元素
    for (int i = 0; i < 1000000; i++) {
        hll.add("user_" + std::to_string(i));
    }

    size_t estimated = hll.estimate();  // ~1000000 (误差<1%)
    std::cout << "Estimated cardinality: " << estimated << "\n";
    std::cout << "Memory usage: " << hll.memory_usage() << " bytes\n";

    // vs std::unordered_set: 需要~40MB
    // HyperLogLog: 只需16KB (节省99.96%)
}
```

```cpp
#include <random>
#include <vector>
#include <unordered_map>

// 简化的LSH实现（余弦相似度）
class LSH {
private:
    struct HashTable {
        std::unordered_map<size_t, std::vector<int>> buckets;
    };

    std::vector<HashTable> tables_;
    std::vector<std::vector<float>> random_vectors_;
    size_t num_tables_;
    size_t num_hashes_;
    size_t dim_;

public:
    LSH(size_t dim, size_t num_tables = 10, size_t num_hashes = 20)
        : dim_(dim), num_tables_(num_tables), num_hashes_(num_hashes) {

        tables_.resize(num_tables_);

        // 生成随机投影向量
        std::random_device rd;
        std::mt19937 gen(rd());
        std::normal_distribution<float> dist(0, 1);

        random_vectors_.resize(num_tables_ * num_hashes_);
        for (auto& vec : random_vectors_) {
            vec.resize(dim_);
            for (size_t i = 0; i < dim_; i++) {
                vec[i] = dist(gen);
            }
        }
    }

    size_t hash_vector(const std::vector<float>& vec, size_t table_id) const {
        size_t hash = 0;

        for (size_t i = 0; i < num_hashes_; i++) {
            const auto& proj = random_vectors_[table_id * num_hashes_ + i];

            // 点积
            float dot = 0;
            for (size_t j = 0; j < dim_; j++) {
                dot += vec[j] * proj[j];
            }

            // 符号位作为哈希位
            if (dot > 0) {
                hash |= (1ULL << i);
            }
        }

        return hash;
    }

    void insert(int id, const std::vector<float>& vec) {
        for (size_t t = 0; t < num_tables_; t++) {
            size_t hash = hash_vector(vec, t);
            tables_[t].buckets[hash].push_back(id);
        }
    }

    std::vector<int> query(const std::vector<float>& vec, size_t k = 10) const {
        std::unordered_set<int> candidates;

        // 收集所有哈希桶中的候选
        for (size_t t = 0; t < num_tables_; t++) {
            size_t hash = hash_vector(vec, t);

            auto it = tables_[t].buckets.find(hash);
            if (it != tables_[t].buckets.end()) {
                for (int id : it->second) {
                    candidates.insert(id);
                }
            }
        }

        // 返回前k个候选（实际应用中需要精确排序）
        std::vector<int> result(candidates.begin(), candidates.end());
        if (result.size() > k) {
            result.resize(k);
        }

        return result;
    }
};

// 使用示例：图像检索
void lsh_example() {
    const size_t DIM = 128;  // 图像特征维度
    LSH lsh(DIM);

    // 插入100万个图像特征向量
    std::vector<std::vector<float>> database(1000000);
    for (int i = 0; i < 1000000; i++) {
        database[i].resize(DIM);
        // 填充特征向量...
        lsh.insert(i, database[i]);
    }

    // 查询
    std::vector<float> query_vec(DIM);
    // 填充查询向量...

    auto start = std::chrono::high_resolution_clock::now();
    auto candidates = lsh.query(query_vec, 10);
    auto lsh_time = std::chrono::high_resolution_clock::now() - start;

    std::cout << "LSH query time: " << lsh_time.count() << " ns\n";

    // vs 线性扫描：需要~100ms
    // LSH: <1ms (快100倍，召回率~90%)
}
```

```cpp
    size_t ef_construction_;  // 构建时的搜索宽度
    size_t max_layers_;

    int entry_point_;

public:
    HNSW(size_t max_m = 16, size_t ef_construction = 200)
        : max_m_(max_m), ef_construction_(ef_construction),
          max_layers_(0), entry_point_(-1) {}

    void insert(const std::vector<float>& vec) {
        int id = nodes_.size();
        Node node;
        node.vector = vec;

        // 确定层数（指数分布）
        size_t layer = random_layer();
        node.neighbors.resize(layer + 1);

        if (entry_point_ == -1) {
            entry_point_ = id;
            max_layers_ = layer;
        } else {
            // 搜索并连接
            std::vector<int> entry_points = {entry_point_};

            for (int lc = max_layers_; lc > static_cast<int>(layer); lc--) {
                entry_points = search_layer(vec, entry_points, 1, lc);

                 lc >= 0; lc--) {
                entry_points = search_layer(vec, entry_points, ef_construction_, lc);

                // 连接到M个最近邻
                size_t m = (lc == 0) ? 2 * max_m_ : max_m_;

            }
        }

        nodes_.push_back(node);

        if (layer > max_layers_) {
            max_layers_ = layer;
            entry_point_ = id;
        }
    }

    std::vector<int> search(const std::vector<float>& query, size_t k) const {
        if (entry_point_ == -1) return {};

        std::vector<int> entry_points = {entry_point_};

        // 从顶层开始
        for (int lc = max_layers_; lc > 0; lc--) {
```

```cpp
// 近似Top-K（使用采样）
template<typename T, typename Compare = std::less<T>>
std::vector<T> approximate_top_k(const std::vector<T>& data,
                                  size_t k,
                                  size_t sample_size = 1000) {
    if (data.size() <= sample_size) {
        // 小数据集：精确排序
        std::vector<T> result = data;
        std::partial_sort(result.begin(), result.begin() + k, result.end(),
                          Compare{});
        result.resize(k);
        return result;
    }

    // 采样
    std::vector<T> sample;
    sample.reserve(sample_size);

    std::random_device rd;
    std::mt19937 gen(rd());
    std::uniform_int_distribution<> dis(0, data.size() - 1);

    for (size_t i = 0; i < sample_size; i++) {
        sample.push_back(data[dis(gen)]);
    }

    // 对采样排序
    std::sort(sample.begin(), sample.end(), Compare{});

    // 估计第k大的值
    T threshold = sample[k * sample_size / data.size()];

    // 过滤并精确排序
    std::vector<T> candidates;
    for (const auto& item : data) {
        if (Compare{}(item, threshold)) {
            candidates.push_back(item);
        }
    }

    if (candidates.size() > k) {
        std::partial_sort(candidates.begin(), candidates.begin() + k,
                          candidates.end(), Compare{});
        candidates.resize(k);
    }

    return candidates;
}
```

### 3.1 Reservoir Sampling（蓄水池采样）

```cpp
// 从流中均匀采样k个元素
template<typename T>
class ReservoirSampler {
private:
    std::vector<T> reservoir_;
    size_t k_;
    size_t count_;
    std::mt19937 gen_;

public:
    explicit ReservoirSampler(size_t k)
        : k_(k), count_(0), gen_(std::random_device{}()) {
        reservoir_.reserve(k);
    }

    void add(const T& item) {
        count_++;

        if (reservoir_.size() < k_) {
            reservoir_.push_back(item);
        } else {
            // 以k/count概率替换
            std::uniform_int_distribution<> dis(0, count_ - 1);
            size_t j = dis(gen_);

            if (j < k_) {
                reservoir_[j] = item;
            }
        }
    }

    const std::vector<T>& get_sample() const {
        return reservoir_;
    }
};

// 使用示例：从无限流中采样
```

```cpp
        : window_size_(window_size) {}

    void add(int value) {
        int idx = data_.size();
        data_.push_back(value);

        // 移除窗口外的元素
        while (!deque_.empty() && deque_.front() <= idx - window_size_) {
            deque_.pop_front();
        }

        // 维护单调性（递减）
        while (!deque_.empty() && data_[deque_.back()] <= value) {
            deque_.pop_back();
        }

        deque_.push_back(idx);
    }

    int get_max() const {
        return data_[deque_.front()];
    }
};

// 使用示例
void sliding_window_example() {
    SlidingWindowMax swm(3);

    std::vector<int> stream = {1, 3, -1, -3, 5, 3, 6, 7};

    for (int val : stream) {
        swm.add(val);
        std::cout << "Max: " << swm.get_max() << "\n";
    }
}
// 时间复杂度：O(1)均摊，O(n)总计
// vs 朴素算法：O(nk)
```

```cpp
class MedianFinder {
private:
    std::priority_queue<int> max_heap_;  // 左半部分（较小）
    std::priority_queue<int, std::vector<int>, std::greater<int>> min_heap_;  // 右半部分（较大）

public:
    void add(int num) {
        // 先加到max_heap
        max_heap_.push(num);

        // 平衡：max_heap最大值 <= min_heap最小值
        min_heap_.push(max_heap_.top());
        max_heap_.pop();

        // 保持size平衡
        if (max_heap_.size() < min_heap_.size()) {
            max_heap_.push(min_heap_.top());
            min_heap_.pop();
        }
    }

    double find_median() {
        if (max_heap_.size() > min_heap_.size()) {
            return max_heap_.top();
        } else {
            return (max_heap_.top() + min_heap_.top()) / 2.0;
        }
    }
};

// 时间复杂度：O(log n)插入，O(1)查询
```

## 第4章 随机化算法

### 4.1 快速选择（QuickSelect）

```cpp
// 第k小元素（平均O(n)）
template<typename T>
T quick_select(std::vector<T>& arr, size_t k) {
    size_t left = 0, right = arr.size() - 1;

    std::random_device rd;
    std::mt19937 gen(rd());

    while (left < right) {
        // 随机选择pivot
        std::uniform_int_distribution<> dis(left, right);
        size_t pivot_idx = dis(gen);
        std::swap(arr[pivot_idx], arr[right]);

        // 分区
        T pivot = arr[right];
        size_t i = left;

        for (size_t j = left; j < right; j++) {
            if (arr[j] < pivot) {
                std::swap(arr[i], arr[j]);
                i++;
            }
        }

        std::swap(arr[i], arr[right]);

        // 递归搜索
        if (i == k) {
            return arr[i];
        } else if (i > k) {
            right = i - 1;
        } else {
            left = i + 1;
        }
    }

    return arr[left];
}

// 性能：平均O(n) vs 排序O(n log n)
```

```cpp
    std::random_device rd;
    std::mt19937 gen(rd());
    std::uniform_real_distribution<> dis(0, 1);

    size_t inside_circle = 0;

        double y = dis(gen);
```

### 4.3 Bloom-Burton算法（随机化最小割）

```cpp
// 图的最小割（随机收缩算法）
class MinCut {
private:
    struct Edge {
        int u, v;
    };

    std::vector<Edge> edges_;
    std::vector<int> parent_;

    int find(int x) {
        if (parent_[x] != x) {
            parent_[x] = find(parent_[x]);
        }
        return parent_[x];
    }

    void unite(int x, int y) {
        parent_[find(x)] = find(y);
    }

public:
    MinCut(const std::vector<Edge>& edges, int n)
        : edges_(edges), parent_(n) {
        for (int i = 0; i < n; i++) {
            parent_[i] = i;
        }
    }

    int compute() {
        std::random_device rd;
        std::mt19937 gen(rd());

        int n = parent_.size();

        // 收缩到2个节点
        while (n > 2) {
            // 随机选择边
            std::uniform_int_distribution<> dis(0, edges_.size() - 1);
            Edge e = edges_[dis(gen)];

            int u = find(e.u);
            int v = find(e.v);

            if (u != v) {
                unite(u, v);
                n--;
            }
        }

        // 计算割边数
        int cut_size = 0;
        for (const auto& e : edges_) {
            if (find(e.u) != find(e.v)) {
                cut_size++;
            }
        }

        return cut_size;
    }
};

// 重复O(n^2 log n)次可获得高概率正确答案
```

```cpp
class BigInt {
public:
    std::vector<int> digits;  // 逆序存储

    BigInt(uint64_t val = 0) {
        if (val == 0) {
            digits = {0};
        } else {
            while (val > 0) {
                digits.push_back(val % 10);
                val /= 10;
            }
        }
    }

    static BigInt karatsuba_multiply(const BigInt& x, const BigInt& y) {
        size_t n = std::max(x.digits.size(), y.digits.size());

        if (n <= 32) {
            // 基础情况：普通乘法
            return naive_multiply(x, y);
        }

        size_t m = n / 2;

        // 分割：x = a*10^m + b, y = c*10^m + d
        BigInt a = x.high(m);
        BigInt b = x.low(m);
        BigInt c = y.high(m);
        BigInt d = y.low(m);

        // 递归计算
        BigInt ac = karatsuba_multiply(a, c);
        BigInt bd = karatsuba_multiply(b, d);
        BigInt abcd = karatsuba_multiply(add(a, b), add(c, d));

        // 结果：ac*10^(2m) + (abcd - ac - bd)*10^m + bd
        BigInt result = shift_left(ac, 2 * m);
        result = add(result, shift_left(subtract(subtract(abcd, ac), bd), m));
        result = add(result, bd);

        return result;
    }

private:
    BigInt high(size_t m) const {
        if (m >= digits.size()) return BigInt(0);
        return BigInt(std::vector<int>(digits.begin() + m, digits.end()));
    }

    BigInt low(size_t m) const {
        size_t len = std::min(m, digits.size());
        return BigInt(std::vector<int>(digits.begin(), digits.begin() + len));
    }

    // ... 辅助函数 ...
};

// 性能：大数乘法快10-100倍
```

#### 5.2.1 四边形不等式优化

```cpp
// 矩阵链乘法（优化到O(n^2)）
int matrix_chain_optimized(const std::vector<int>& dims) {
    int n = dims.size() - 1;

    std::vector<std::vector<int>> dp(n, std::vector<int>(n, 0));
    std::vector<std::vector<int>> split(n, std::vector<int>(n, 0));

    // 枚举链长度
    for (int len = 2; len <= n; len++) {
        for (int i = 0; i + len - 1 < n; i++) {
            int j = i + len - 1;
            dp[i][j] = std::numeric_limits<int>::max();

            // 利用四边形不等式：split[i][j-1] <= split[i][j] <= split[i+1][j]
            int start = (i == 0) ? 0 : split[i][j-1];
            int end = (j == n-1) ? n-1 : split[i+1][j];

            for (int k = start; k <= end; k++) {
                int cost = dp[i][k] + dp[k+1][j] +
                           dims[i] * dims[k+1] * dims[j+1];

                if (cost < dp[i][j]) {
                    dp[i][j] = cost;
                    split[i][j] = k;
                }
            }
        }
    }

    return dp[0][n-1];
}

// 性能：O(n^2) vs 朴素O(n^3)
```

#### 5.2.2 斜率优化DP

```cpp
// 凸包优化（CHT - Convex Hull Trick）
class ConvexHullTrick {
private:
    struct Line {
        int64_t m, c;  // y = mx + c

        int64_t eval(int64_t x) const {
            return m * x + c;
        }

        // 交点x坐标
        double intersect(const Line& other) const {
            return static_cast<double>(other.c - c) / (m - other.m);
        }
    };

    std::deque<Line> hull_;

public:
    void add_line(int64_t m, int64_t c) {
        Line new_line{m, c};

        // 移除无用的线
        while (hull_.size() >= 2) {
            Line& last = hull_.back();
            Line& second_last = hull_[hull_.size() - 2];

            if (second_last.intersect(last) >= second_last.intersect(new_line)) {
                hull_.pop_back();
            } else {
                break;
            }
        }

        hull_.push_back(new_line);
    }

    int64_t query(int64_t x) {
        // 移除过时的线
        while (hull_.size() >= 2 && hull_[0].eval(x) >= hull_[1].eval(x)) {
            hull_.pop_front();
        }

        return hull_[0].eval(x);
    }
};

// 使用示例：最小费用问题
int64_t solve_with_cht(const std::vector<int64_t>& costs) {
    int n = costs.size();
    std::vector<int64_t> dp(n + 1, 0);

    ConvexHullTrick cht;
    cht.add_line(0, 0);

    for (int i = 1; i <= n; i++) {
        dp[i] = cht.query(i) + costs[i-1];
        cht.add_line(-2 * i, dp[i] + i * i);
    }

    return dp[n];
}

// 性能：O(n) vs 朴素O(n^2)
```

```cpp
// 使用优先队列的Dijkstra（O((V+E) log V)）
std::vector<int> dijkstra_optimized(const Graph& graph, int source) {
    int n = graph.num_vertices();
    std::vector<int> dist(n, std::numeric_limits<int>::max());
    std::vector<bool> visited(n, false);

    using PII = std::pair<int, int>;  // (distance, vertex)
    std::priority_queue<PII, std::vector<PII>, std::greater<PII>> pq;

    dist[source] = 0;
    pq.push({0, source});

    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();

        if (visited[u]) continue;
        visited[u] = true;

        for (auto [v, weight] : graph.adj[u]) {
            if (dist[u] + weight < dist[v]) {
                dist[v] = dist[u] + weight;
                pq.push({dist[v], v});
            }
        }
    }

    return dist;
}

// vs 朴素Dijkstra: O(V^2)
// 优化：50-1000倍加速（稀疏图）
```

```cpp
// KMP字符串匹配（O(n+m)）
class KMP {
private:
    std::vector<int> lps_;  // Longest Proper Prefix which is also Suffix

    void compute_lps(const std::string& pattern) {
        int m = pattern.size();
        lps_.resize(m, 0);

        int len = 0;
        int i = 1;

        while (i < m) {
            if (pattern[i] == pattern[len]) {
                len++;
                lps_[i] = len;
                i++;
            } else {
                if (len != 0) {
                    len = lps_[len - 1];
                } else {
                    lps_[i] = 0;
                    i++;
                }
            }
        }
    }

public:
    std::vector<int> search(const std::string& text, const std::string& pattern) {
        compute_lps(pattern);

        std::vector<int> matches;
        int n = text.size();
        int m = pattern.size();

        int i = 0;  // text索引
        int j = 0;  // pattern索引

        while (i < n) {
            if (text[i] == pattern[j]) {
                i++;
                j++;
            }

            if (j == m) {
                matches.push_back(i - j);
                j = lps_[j - 1];
            } else if (i < n && text[i] != pattern[j]) {
                if (j != 0) {
                    j = lps_[j - 1];
                } else {
                    i++;
                }
            }
        }

        return matches;
    }
};

// vs 朴素算法：O(nm)
// 性能：10-100倍加速
```

## 总结

高级算法优化的核心思想：

1. **概率换空间**：Bloom Filter、HyperLogLog等牺牲少量准确性换取巨大空间节省
2. **近似换速度**：LSH、近似Top-K等牺牲少量精度换取数量级加速
3. **随机化**：QuickSelect、蒙特卡洛等利用随机性简化问题

5. **数学优化**：四边形不等式、斜率优化等降低复杂度

记住：**没有银弹，选择合适的算法权衡准确性、速度、空间！** 🎯
