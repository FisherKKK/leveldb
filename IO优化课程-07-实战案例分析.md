## 1. LevelDB IO优化深度剖析

### 1.1 LevelDB架构概览

```
LevelDB IO层次：

写入路径：
Application
    ↓
WriteBatch (批量)
    ↓
WAL (顺序写) ───────┐
    ↓               │ (持久化)
MemTable (内存)     │
    ↓               │
Immutable MemTable  │
    ↓               ↓
SSTable Level-0 (后台flush)
    ↓
Compaction (后台merge)
    ↓
SSTable Level-1..N

读取路径：
Application
    ↓
MemTable (check)
    ↓
Immutable MemTable (check)
    ↓
Level-0 SSTables (可能重叠，需要查所有)
    ↓
Level-1+ SSTables (二分查找)
    ↓
Block Cache (LRU)
    ↓
Bloom Filter (快速排除)
```

### 1.2 LevelDB的IO优化技术

#### 1.2.1 写入优化

**1. Write-Ahead Log (WAL) 顺序写：**

```cpp
// db/log_writer.cc
Status Writer::AddRecord(const Slice& slice) {
  const char* ptr = slice.data();
  size_t left = slice.size();

  Status s;
  bool begin = true;
  do {
    const int leftover = kBlockSize - block_offset_;

    // 批量写入，减少系统调用
    if (leftover < kHeaderSize) {
      // 填充到block边界
      dest_->Append(Slice("\x00\x00\x00\x00\x00\x00", leftover));
      block_offset_ = 0;
    }

    const size_t avail = kBlockSize - block_offset_ - kHeaderSize;
    const size_t fragment_length = (left < avail) ? left : avail;

    // 构造header
    char buf[kHeaderSize];
    buf[4] = static_cast<char>(fragment_length & 0xff);
    buf[5] = static_cast<char>(fragment_length >> 8);
    buf[6] = type;

    // CRC校验
    uint32_t crc = crc32c::Extend(type_crc_[type], ptr, fragment_length);
    crc = crc32c::Mask(crc);
    EncodeFixed32(buf, crc);

    // 批量写入：header + data
    s = dest_->Append(Slice(buf, kHeaderSize));
    if (s.ok()) {
      s = dest_->Append(Slice(ptr, fragment_length));
      if (s.ok()) {
        s = dest_->Flush();  // 刷到OS buffer
      }
    }

    ptr += fragment_length;
    left -= fragment_length;
    begin = false;
  } while (s.ok() && left > 0);

  return s;
}
```

**优化点**：
- 顺序写入日志文件（HDD友好）
- 批量写入减少系统调用
- 32KB块对齐（文件系统友好）
- 异步刷盘选项（WriteOptions::sync）

**2. WriteBatch 批量写入：**

```cpp
// db/write_batch.cc
void WriteBatch::Put(const Slice& key, const Slice& value) {
  WriteBatchInternal::SetCount(this, WriteBatchInternal::Count(this) + 1);
  rep_.push_back(static_cast<char>(kTypeValue));
  PutLengthPrefixedSlice(&rep_, key);    // 内存拷贝
  PutLengthPrefixedSlice(&rep_, value);  // 内存拷贝
}

// db/db_impl.cc - 批量提交
Status DBImpl::Write(const WriteOptions& options, WriteBatch* updates) {
  Writer w(&mutex_);
  w.batch = updates;
  w.sync = options.sync;
  w.done = false;

  MutexLock l(&mutex_);
  writers_.push_back(&w);

  // 等待前面的写入完成或成为leader
  while (!w.done && &w != writers_.front()) {
    w.cv.Wait();
  }

  if (w.done) {
    return w.status;
  }

  // Group commit: 合并多个WriteBatch
  WriteBatch* updates = BuildBatchGroup(&last_writer);

  // 写WAL（一次系统调用）
  status = log_->AddRecord(WriteBatchInternal::Contents(updates));

  // 写MemTable（内存操作）
  if (status.ok()) {
    status = WriteBatchInternal::InsertInto(updates, mem_);
  }

  // 唤醒其他等待的writer
  while (true) {
    Writer* ready = writers_.front();
    writers_.pop_front();
    if (ready != &w) {
      ready->status = status;
      ready->done = true;
      ready->cv.Signal();
    }
    if (ready == last_writer) break;
  }

  if (!writers_.empty()) {
    writers_.front()->cv.Signal();  // 唤醒下一个leader
  }

  return status;
}
```

**优化点**：
- Group Commit：多个写请求合并为一次磁盘IO
- 减少WAL写入次数
- 减少fsync次数（可配置）
- 写入延迟换取吞吐量

#### 1.2.2 读取优化

**1. Block Cache（LRU缓存）：**

```cpp
// table/table.cc
Iterator* Table::BlockReader(void* arg, const ReadOptions& options,
                              const Slice& index_value) {
  Table* table = reinterpret_cast<Table*>(arg);
  Cache* block_cache = table->rep_->options.block_cache;
  Block* block = nullptr;
  Cache::Handle* cache_handle = nullptr;

  BlockHandle handle;
  Slice input = index_value;
  Status s = handle.DecodeFrom(&input);

  if (s.ok()) {
    BlockContents contents;
    if (block_cache != nullptr) {
      // 1. 先查cache
      char cache_key_buffer[16];
      EncodeFixed64(cache_key_buffer, table->rep_->cache_id);
      EncodeFixed64(cache_key_buffer + 8, handle.offset());
      Slice key(cache_key_buffer, sizeof(cache_key_buffer));

      cache_handle = block_cache->Lookup(key);
      if (cache_handle != nullptr) {
        // Cache hit
        block = reinterpret_cast<Block*>(
            block_cache->Value(cache_handle));
      } else {
        // Cache miss: 从文件读取
        s = ReadBlock(table->rep_->file, options, handle, &contents);
        if (s.ok()) {
          block = new Block(contents);
          // 插入cache
          cache_handle = block_cache->Insert(
              key, block, block->size(), &DeleteCachedBlock);
        }
      }
    }
  }

  Iterator* iter;
  if (block != nullptr) {
    iter = block->NewIterator(table->rep_->options.comparator);
    if (cache_handle == nullptr) {
      iter->RegisterCleanup(&DeleteBlock, block, nullptr);
    } else {
      iter->RegisterCleanup(&ReleaseBlock, block_cache, cache_handle);
    }
  }
  return iter;
}
```

**优化点**：
- LRU缓存热数据块
- 缓存key = (file_id, block_offset)
- 减少磁盘读取
- 可配置缓存大小

**2. Bloom Filter快速过滤：**

```cpp
// table/filter_block.cc
bool FilterBlockReader::KeyMayMatch(uint64_t block_offset,
                                     const Slice& key) {
  uint64_t index = block_offset >> base_lg_;
  if (index < num_) {
    uint32_t start = DecodeFixed32(offset_ + index * 4);
    uint32_t limit = DecodeFixed32(offset_ + index * 4 + 4);
    if (start <= limit && limit <= static_cast<size_t>(offset_ - data_)) {
      Slice filter = Slice(data_ + start, limit - start);
      // 使用Bloom Filter检查
      return policy_->KeyMayMatch(key, filter);
    }
  }
  return true;  // 假阳性：可能存在
}

// util/bloom.cc - Bloom Filter实现
class BloomFilterPolicy : public FilterPolicy {
 public:
  explicit BloomFilterPolicy(int bits_per_key) : bits_per_key_(bits_per_key) {
    k_ = static_cast<size_t>(bits_per_key * 0.69);  // ln(2)
    if (k_ < 1) k_ = 1;
    if (k_ > 30) k_ = 30;
  }

  void CreateFilter(const Slice* keys, int n, std::string* dst) const override {
    size_t bits = n * bits_per_key_;
    if (bits < 64) bits = 64;
    size_t bytes = (bits + 7) / 8;
    bits = bytes * 8;

    const size_t init_size = dst->size();
    dst->resize(init_size + bytes, 0);
    dst->push_back(static_cast<char>(k_));
    char* array = &(*dst)[init_size];

    // 对每个key计算k个hash
    for (int i = 0; i < n; i++) {
      uint32_t h = BloomHash(keys[i]);
      const uint32_t delta = (h >> 17) | (h << 15);
      for (size_t j = 0; j < k_; j++) {
        const uint32_t bitpos = h % bits;
        array[bitpos / 8] |= (1 << (bitpos % 8));  // 设置bit
        h += delta;
      }
    }
  }

  bool KeyMayMatch(const Slice& key, const Slice& bloom_filter) const override {
    const size_t len = bloom_filter.size();
    if (len < 2) return false;

    const char* array = bloom_filter.data();
    const size_t bits = (len - 1) * 8;
    const size_t k = array[len - 1];
    if (k > 30) return true;

    uint32_t h = BloomHash(key);
    const uint32_t delta = (h >> 17) | (h << 15);
    for (size_t j = 0; j < k; j++) {
      const uint32_t bitpos = h % bits;
      if ((array[bitpos / 8] & (1 << (bitpos % 8))) == 0) {
        return false;  // 确定不存在
      }
      h += delta;
    }
    return true;  // 可能存在
  }
};
```

**优化点**：
- 每个SSTable block有独立的Bloom Filter
- 快速判断key是否可能存在
- 减少不必要的磁盘读取
- 假阳性率可配置（默认10 bits/key，1%误报率）

#### 1.2.3 Compaction优化

```cpp
// db/db_impl.cc
void DBImpl::BackgroundCompaction() {
  mutex_.AssertHeld();

  if (imm_ != nullptr) {
    // Minor compaction: MemTable → Level-0 SSTable
    CompactMemTable();
    return;
  }

  Compaction* c;
  bool is_manual = (manual_compaction_ != nullptr);

  if (is_manual) {
    ManualCompaction* m = manual_compaction_;
    c = versions_->CompactRange(m->level, m->begin, m->end);
  } else {
    // 自动选择需要compaction的level
    c = versions_->PickCompaction();
  }

  Status status;
  if (c == nullptr) {
    // Nothing to do
  } else if (!is_manual && c->IsTrivialMove()) {
    // 优化：如果只是简单移动文件，不需要merge
    // Move file to next level
    FileMetaData* f = c->input(0, 0);
    c->edit()->RemoveFile(c->level(), f->number);
    c->edit()->AddFile(c->level() + 1, f->number, f->file_size,
                       f->smallest, f->largest);
    status = versions_->LogAndApply(c->edit(), &mutex_);
  } else {
    // 真正的compaction: merge多个SSTable
    CompactionState* compact = new CompactionState(c);
    status = DoCompactionWork(compact);
    CleanupCompaction(compact);
    c->ReleaseInputs();
    RemoveObsoleteFiles();
  }

  delete c;
}

Status DBImpl::DoCompactionWork(CompactionState* compact) {
  // 创建多路归并迭代器
  Iterator* input = versions_->MakeInputIterator(compact->compaction);

  input->SeekToFirst();
  Status status;
  ParsedInternalKey ikey;
  std::string current_user_key;
  bool has_current_user_key = false;
  SequenceNumber last_sequence_for_key = kMaxSequenceNumber;

  // 归并排序，同时去除重复和删除的key
  while (input->Valid() && !shutting_down_.load(std::memory_order_acquire)) {
    // 优先处理immutable memtable（减少内存压力）
    if (has_imm_.load(std::memory_order_relaxed)) {
      mutex_.Lock();
      if (imm_ != nullptr) {
        CompactMemTable();
        background_work_finished_signal_.SignalAll();
      }
      mutex_.Unlock();
    }

    Slice key = input->key();

    // 如果输出文件太大，切分文件
    if (compact->builder != nullptr &&
        compact->builder->FileSize() >= compact->compaction->MaxOutputFileSize()) {
      status = FinishCompactionOutputFile(compact, input);
    }

    // 解析internal key
    if (!ParseInternalKey(key, &ikey)) {
      current_user_key.clear();
      has_current_user_key = false;
      last_sequence_for_key = kMaxSequenceNumber;
    } else {
      if (!has_current_user_key ||
          user_comparator()->Compare(ikey.user_key, Slice(current_user_key)) != 0) {
        // 新的user key
        current_user_key.assign(ikey.user_key.data(), ikey.user_key.size());
        has_current_user_key = true;
        last_sequence_for_key = kMaxSequenceNumber;
      }

      // 决定是否保留这个key
      bool drop = false;
      if (last_sequence_for_key <= compact->smallest_snapshot) {
        // 旧版本，可以丢弃
        drop = true;
      } else if (ikey.type == kTypeDeletion &&
                 ikey.sequence <= compact->smallest_snapshot &&
                 compact->compaction->IsBaseLevelForKey(ikey.user_key)) {
        // 删除标记可以丢弃
        drop = true;
      }

      last_sequence_for_key = ikey.sequence;

      if (!drop) {
        // 写入输出文件
        if (compact->builder == nullptr) {
          status = OpenCompactionOutputFile(compact);
        }
        compact->builder->Add(key, input->value());

        // 更新统计
        compact->current_output()->smallest_seqno =
            std::min(compact->current_output()->smallest_seqno, ikey.sequence);
        compact->current_output()->largest_seqno =
            std::max(compact->current_output()->largest_seqno, ikey.sequence);
      }
    }

    input->Next();
  }

  if (status.ok() && compact->builder != nullptr) {
    status = FinishCompactionOutputFile(compact, input);
  }

  CompactionStats stats;
  stats.micros = env_->NowMicros() - start_micros - imm_micros;
  stats.bytes_read = compact->compaction->input_bytes();
  stats.bytes_written = compact->total_bytes;

  mutex_.Lock();
  stats_[compact->compaction->level() + 1].Add(stats);
  mutex_.Unlock();

  return status;
}
```

**Compaction优化点**：
1. **后台异步执行**：不阻塞前台读写
2. **智能触发**：基于大小和文件数量
3. **Trivial Move**：无重叠时直接移动文件
4. **多路归并**：高效合并多个SSTable
5. **去重和删除标记清理**：减少存储空间
6. **限流**：避免影响前台性能

### 6.1 案例：电商商品详情页读优化

**背景**：

- 平均响应时间：200ms（P99: 500ms，不满足50ms要求）

**问题分析**：

```bash
# 发现大量SELECT查询阻塞

# 2. 分析查询性能
mysql> EXPLAIN SELECT * FROM products WHERE product_id = 12345;
# type: ALL（全表扫描！）
# rows: 10,000,000（扫描1000万行）

# 3. 查看索引
mysql> SHOW INDEX FROM products;
# 没有product_id索引！
```

**优化方案**：

```sql
-- 优化1：添加索引（读取加速100倍）
ALTER TABLE products ADD INDEX idx_product_id (product_id);

-- 优化2：覆盖索引（避免回表）
-- Bad: SELECT * 需要回表读取所有字段
SELECT * FROM products WHERE product_id = 12345;

-- Good: 只查需要的字段，建立覆盖索引
ALTER TABLE products ADD INDEX idx_product_cover
  (product_id, name, price, stock, description);

SELECT name, price, stock, description
FROM products
WHERE product_id = 12345;
-- 直接从索引获取数据，无需回表！

-- 优化3：多级缓存
-- L1: Redis热点商品缓存（10万个热门商品）
-- L2: 本地内存缓存（LRU，1万个）
-- L3: MySQL数据库
```

**代码实现**：

```java
@Service
public class ProductService {
    @Autowired
    private RedisTemplate<String, Product> redisTemplate;

    @Autowired
    private ProductMapper productMapper;

    // 本地缓存
    private LoadingCache<Long, Product> localCache = CacheBuilder.newBuilder()
        .maximumSize(10000)
        .expireAfterWrite(5, TimeUnit.MINUTES)
        .build(new CacheLoader<Long, Product>() {
            @Override
            public Product load(Long productId) throws Exception {
                return loadFromRedisOrDB(productId);
            }
        });

    public Product getProduct(Long productId) {
        // L1: 本地缓存（最快，1μs）
        try {
            return localCache.get(productId);
        } catch (Exception e) {
            return loadFromRedisOrDB(productId);
        }
    }

    private Product loadFromRedisOrDB(Long productId) {
        // L2: Redis缓存（快，1ms）
        String key = "product:" + productId;
        Product product = redisTemplate.opsForValue().get(key);

        if (product != null) {
            return product;
        }

        // L3: 数据库查询（慢，10ms）
        product = productMapper.selectByPrimaryKey(productId);

        if (product != null) {
            // 写入Redis缓存
            redisTemplate.opsForValue().set(key, product, 1, TimeUnit.HOURS);
        }

        return product;
    }
}
```

**优化效果**：

```
优化前：
- P50延迟：200ms
- P99延迟：500ms
- QPS：10,000（数据库瓶颈）
- 数据库CPU：90%

优化后：
- P50延迟：2ms（本地缓存命中率：70%）
- P99延迟：15ms（Redis缓存命中率：95%）
- QPS：100,000（10倍提升）
- 数据库CPU：10%（命中率99%时仅1%请求到DB）

读取路径分布：
- 本地缓存：70%（1μs）
- Redis缓存：29%（1ms）
- 数据库查询：1%（10ms）
```

```python
class FeatureStore:
    def __init__(self):
        # 所有用户特征加载到内存
        self.user_features = {}  # user_id -> numpy.array

        # 热门物品特征加载到内存
        self.hot_item_features = {}  # item_id -> numpy.array

        # 冷门物品特征在Redis

        # 内存查找（10ns）
        return self.user_features.get(user_id)

        # 先查内存
        if item_id in self.hot_item_features:
            return self.hot_item_features[item_id]  # 10ns

        # 再查Redis
        return self.redis_client.hgetall(f"item:{item_id}")  # 1ms

        redis_items = []

        for item_id in item_ids:

            else:
                redis_items.append(item_id)

            pipeline = self.redis_client.pipeline()
            for item_id in redis_items:
                pipeline.hgetall(f"item:{item_id}")
            redis_features = pipeline.execute()
            features.extend(redis_features)

        return features
```

```cpp
    std::vector<int64_t> timestamps_;
    std::vector<float> temperatures_;
    std::vector<float> humidities_;
    std::vector<float> pressures_;

    std::vector<std::string> locations_;

        std::vector<float> results;

        // 1. 二分查找起始位置
        auto start_it = std::lower_bound(
            timestamps_.begin(), timestamps_.end(), start_time);

        // 2. 顺序扫描（连续内存，cache友好）
        size_t start_idx = std::distance(timestamps_.begin(), start_it);

        for (size_t i = start_idx; i < timestamps_.size(); i++) {
            if (timestamps_[i] > end_time) {
                break;
            }

            results.push_back(temperatures_[i]);
            // 优势：只读取需要的列！
        }

        return results;
    }

    // 范围聚合查询（列存储天然优势）
    struct AggregateResult {
        float avg_temp;
        float min_temp;
        float max_temp;
    };

    AggregateResult AggregateTemperature(int64_t start_time, int64_t end_time) {
        auto start_it = std::lower_bound(
            timestamps_.begin(), timestamps_.end(), start_time);
        size_t start_idx = std::distance(timestamps_.begin(), start_it);

        float sum = 0, min_val = FLT_MAX, max_val = FLT_MIN;
        size_t count = 0;

        // SIMD友好：连续float数组
        for (size_t i = start_idx; i < timestamps_.size(); i++) {
            if (timestamps_[i] > end_time) break;

            float temp = temperatures_[i];
            sum += temp;
            min_val = std::min(min_val, temp);
            max_val = std::max(max_val, temp);
            count++;
        }

        return {sum / count, min_val, max_val};
    }
};
```

```cpp
class CompressedColumnStore {
    std::vector<int64_t> timestamps_;  // Delta编码
    std::vector<uint8_t> compressed_temperatures_;  // Snappy压缩

public:
    void Insert(int64_t timestamp, float temperature) {
        // 1. Timestamp delta编码
        if (!timestamps_.empty()) {
            int64_t delta = timestamp - timestamps_.back();
            timestamps_.push_back(delta);  // 存储差值（更小）
        } else {
            timestamps_.push_back(timestamp);
        }

        // 2. Temperature批量压缩
        // 积累一批数据后压缩
        temp_buffer_.push_back(temperature);

        if (temp_buffer_.size() >= 1024) {
            CompressAndStore();
        }
    }

private:
    std::vector<float> temp_buffer_;

    void CompressAndStore() {
        // 使用Snappy压缩
        size_t compressed_size = snappy::MaxCompressedLength(
            temp_buffer_.size() * sizeof(float));

        std::vector<uint8_t> compressed(compressed_size);
        snappy::RawCompress(
            reinterpret_cast<const char*>(temp_buffer_.data()),
            temp_buffer_.size() * sizeof(float),
            reinterpret_cast<char*>(compressed.data()),
            &compressed_size);

        compressed.resize(compressed_size);
        compressed_temperatures_.insert(
            compressed_temperatures_.end(),
            compressed.begin(),
            compressed.end());

        temp_buffer_.clear();
    }
};
```

```cpp
    float x, y, z;
};

struct VelocityComponent {
    float vx, vy, vz;
};

struct RenderComponent {
    std::string name;
    Texture* texture;
    Model* model;
};

    // 所有位置连续存储（cache友好）
    std::vector<PositionComponent> positions_;

    // 所有速度连续存储
    std::vector<VelocityComponent> velocities_;

    // 渲染组件分开存储（冷数据）
    std::vector<RenderComponent> render_components_;

    // 实体到组件的映射
    std::unordered_map<int, size_t> entity_to_component_;

public:
    void UpdatePositions(float dt) {
        // 顺序访问连续内存（极高cache命中率）
        for (size_t i = 0; i < positions_.size(); i++) {
            positions_[i].x += velocities_[i].vx * dt;
            positions_[i].y += velocities_[i].vy * dt;
            positions_[i].z += velocities_[i].vz * dt;
        }

        // 优势：
        // - 连续内存：顺序访问
        // - 紧凑数据：每个实体24字节
        // - Cache利用率：~100%
        // - 编译器自动向量化（SIMD）
    }

    // SIMD优化版本
    void UpdatePositionsSIMD(float dt) {
        #ifdef __AVX__
        __m256 dt_vec = _mm256_set1_ps(dt);

        // 一次处理8个实体（AVX）
        for (size_t i = 0; i < positions_.size(); i += 8) {
            // 加载8个x坐标
            __m256 x = _mm256_loadu_ps(&positions_[i].x);
            __m256 vx = _mm256_loadu_ps(&velocities_[i].vx);

            // 向量化计算：x += vx * dt（8个并行）
            x = _mm256_fmadd_ps(vx, dt_vec, x);

            // 存储结果
            _mm256_storeu_ps(&positions_[i].x, x);

            // 同样处理y, z坐标...
        }
        #endif
    }
```

```cpp
// 好：紧凑数组 + 价格索引
class OrderBook_Optimized {
    // 订单数据：连续存储
    struct alignas(64) Order {  // Cache line对齐
        int64_t order_id;
        double price;
        int64_t quantity;
        uint32_t next_index;  // 索引而非指针
        char padding[28];     // 填充到64字节
    };

    // 预分配订单池（避免动态分配）
    std::array<Order, 100000> order_pool_;
    size_t pool_index_ = 0;

    // 价格级别索引：快速查找
    std::map<double, std::vector<uint32_t>> price_to_orders_;

    // 最优价格缓存
    double best_bid_ = 0;
    double best_ask_ = std::numeric_limits<double>::max();

public:
    void InsertBuyOrder(double price, int64_t quantity) {
        // 从预分配池获取（无动态分配）
        uint32_t order_idx = pool_index_++;
        Order& order = order_pool_[order_idx];

        order.order_id = next_id_++;
        order.price = price;
        order.quantity = quantity;

        // 更新价格索引
        price_to_orders_[price].push_back(order_idx);

        // 更新最优价格缓存
        if (price > best_bid_) {
            best_bid_ = price;
        }
    }

    // O(1)获取最优价格（缓存）
    double GetBestBid() const {
        return best_bid_;
    }

    // 订单匹配（连续内存访问）
    std::vector<Trade> MatchOrders() {
        std::vector<Trade> trades;

        // 获取最优买价和卖价的订单
        auto& bid_orders = price_to_orders_[best_bid_];
        auto& ask_orders = price_to_orders_[best_ask_];

        // 连续内存访问（cache友好）
        for (uint32_t bid_idx : bid_orders) {
            Order& bid = order_pool_[bid_idx];

            for (uint32_t ask_idx : ask_orders) {
                Order& ask = order_pool_[ask_idx];

                if (bid.price >= ask.price) {
                    int64_t trade_qty = std::min(bid.quantity, ask.quantity);

                    trades.push_back({bid.order_id, ask.order_id,
                                      ask.price, trade_qty});

                    bid.quantity -= trade_qty;
                    ask.quantity -= trade_qty;

                    if (bid.quantity == 0) break;
                }
            }
        }

        return trades;
    }

private:
    int64_t next_id_ = 0;

    struct Trade {
        int64_t bid_order_id;
        int64_t ask_order_id;
        double price;
        int64_t quantity;
    };
};
```

```cpp
    // Cache line分离（避免false sharing）
    struct alignas(64) PriceLevel {
        std::atomic<int64_t> total_quantity{0};
        std::atomic<uint32_t> order_count{0};

        // 同一价格的订单索引
        std::array<std::atomic<uint32_t>, 1000> order_indices;
    };

    // 买卖价格级别（分离到不同cache line）
    alignas(64) std::array<PriceLevel, 10000> buy_levels_;
    alignas(64) std::array<PriceLevel, 10000> sell_levels_;

    // 订单池（cache line对齐）
    alignas(64) std::array<Order, 1000000> order_pool_;
    std::atomic<size_t> pool_index_{0};

public:
    void InsertBuyOrder(double price, int64_t quantity) {
        // 无锁分配订单
        size_t order_idx = pool_index_.fetch_add(1, std::memory_order_relaxed);

        Order& order = order_pool_[order_idx];
        order.order_id = next_id_.fetch_add(1, std::memory_order_relaxed);
        order.price = price;
        order.quantity = quantity;

        // 计算价格级别
        size_t level = static_cast<size_t>(price * 100);  // 假设价格精度0.01

        PriceLevel& pl = buy_levels_[level];

        // 无锁更新
        pl.total_quantity.fetch_add(quantity, std::memory_order_release);
        uint32_t count = pl.order_count.fetch_add(1, std::memory_order_release);
        pl.order_indices[count].store(order_idx, std::memory_order_release);
    }

private:
    std::atomic<int64_t> next_id_{0};
};
```

### 7.4 案例：机器学习特征存储优化

**背景**：

**原始实现（行存储）：**

```cpp
    int64_t user_id;
    float features[1000];  // 1000维特征
};  // 4004 bytes per user

class FeatureStore_Row {
    std::unordered_map<int64_t, UserFeatures_Row> features_;

public:
    // 批量获取特征（用于模型推理）
    std::vector<std::vector<float>> BatchGetFeatures(
        const std::vector<int64_t>& user_ids) {

        std::vector<std::vector<float>> results;

        for (int64_t uid : user_ids) {
            auto it = features_.find(uid);
            if (it != features_.end()) {
                results.emplace_back(
                    it->second.features,
                    it->second.features + 1000);
            }
        }

        return results;
    }
};

// 问题：
// - Hash查找开销
// - 特征分散，cache miss多
// - 无法SIMD优化
```

**优化实现（列存储 + SIMD）：**

```cpp
// 列式存储：每一维特征连续存储
class FeatureStore_Column {
    size_t num_users_;
    static constexpr size_t NUM_FEATURES = 1000;

    // 每一维特征独立存储（连续数组）
    std::array<std::vector<float>, NUM_FEATURES> feature_columns_;

    // 用户ID到索引的映射
    std::unordered_map<int64_t, size_t> user_to_index_;

public:
    FeatureStore_Column(size_t num_users) : num_users_(num_users) {
        for (auto& col : feature_columns_) {
            col.resize(num_users, 0.0f);
        }
    }

    // 插入特征
    void InsertFeatures(int64_t user_id, const std::vector<float>& features) {
        size_t idx = user_to_index_.size();
        user_to_index_[user_id] = idx;

        for (size_t f = 0; f < NUM_FEATURES; f++) {
            feature_columns_[f][idx] = features[f];
        }
    }

    // 批量获取特征（优化版本）
    void BatchGetFeaturesSIMD(
        const std::vector<int64_t>& user_ids,
        float* output) {  // 输出: [user_count × 1000]

        std::vector<size_t> indices;
        indices.reserve(user_ids.size());

        for (int64_t uid : user_ids) {
            auto it = user_to_index_.find(uid);
            if (it != user_to_index_.end()) {
                indices.push_back(it->second);
            }
        }

        // 转置：从列存储转为行存储（供模型使用）
        // 使用SIMD加速
        #ifdef __AVX__
        for (size_t f = 0; f < NUM_FEATURES; f++) {
            const float* feature_col = feature_columns_[f].data();

            for (size_t i = 0; i < indices.size(); i += 8) {
                // 一次处理8个用户
                __m256 values = _mm256_set_ps(
                    feature_col[indices[i]],
                    feature_col[indices[i+1]],
                    feature_col[indices[i+2]],
                    feature_col[indices[i+3]],
                    feature_col[indices[i+4]],
                    feature_col[indices[i+5]],
                    feature_col[indices[i+6]],
                    feature_col[indices[i+7]]
                );

                _mm256_storeu_ps(output + i * NUM_FEATURES + f, values);
            }
        }
        #endif
    }

    // 特征归一化（列存储天然优势）
    void NormalizeFeature(size_t feature_idx) {
        auto& col = feature_columns_[feature_idx];

        // 计算均值和标准差（顺序访问，cache友好）
        float sum = 0, sum_sq = 0;

        for (float val : col) {
            sum += val;
            sum_sq += val * val;
        }

        float mean = sum / num_users_;
        float std = std::sqrt(sum_sq / num_users_ - mean * mean);

        // 归一化（SIMD加速）
        #ifdef __AVX__
        __m256 mean_vec = _mm256_set1_ps(mean);
        __m256 std_vec = _mm256_set1_ps(std);

        for (size_t i = 0; i < num_users_; i += 8) {
            __m256 vals = _mm256_loadu_ps(&col[i]);
            vals = _mm256_sub_ps(vals, mean_vec);
            vals = _mm256_div_ps(vals, std_vec);
            _mm256_storeu_ps(&col[i], vals);
        }
        #endif
    }
};
```

```java
        } else if (result == 0) {
            return SecKillResult.fail("库存不足");
        }

        // 2. 发送消息到Kafka（异步处理）
        OrderMessage message = new OrderMessage(productId, userId);
        kafka.send("order-topic", JSON.toJSONString(message));

        return SecKillResult.success();
    }
}

// 消费者（异步处理订单）
@KafkaListener(topics = "order-topic", concurrency = "10")
public void processOrder(String message) {
    OrderMessage order = JSON.parseObject(message, OrderMessage.class);

    try {
        // 1. 创建订单
        Order orderEntity = createOrder(order);

        // 2. 扣减库存（数据库最终一致性）
        productMapper.decreaseStock(order.getProductId(), 1);

        // 3. 保存订单
        orderMapper.insert(orderEntity);

    } catch (Exception e) {
        // 失败：库存回滚
        String stockKey = "seckill:stock:" + order.getProductId();
        redis.opsForValue().increment(stockKey, 1);

        // 记录失败日志
        log.error("订单处理失败", e);
    }
}
```
